#!/usr/bin/env bash
# Throwaway browser check for Laysi/trustiot-v2#1147, run from the root of a copy of the worktree.
# Serves the UI preview (fake device API, in the page) with the fake exposed to the script, and drives
# it in the pinned Playwright image with a controllable page clock.
set -uo pipefail
root=$PWD
work=$root/.e2e-1147-work
rm -rf "$work"
mkdir -p "$work/pw" "$work/out"
bun install --cwd frontend --frozen-lockfile > "$work/install.log" 2>&1 || { tail -20 "$work/install.log"; exit 1; }
bun .e2e-1147-src/patch.ts
cp -rL frontend/node_modules/playwright-core "$work/pw/"
(cd frontend && bun --bun vite --mode ui-preview --host 127.0.0.1 --port 5174 > "$work/vite.log" 2>&1) &
vite=$!
for _ in $(seq 1 120); do curl -fsS -o /dev/null http://127.0.0.1:5174/ && break; sleep 1; done
image=$(bun -e 'import { PLAYWRIGHT_IMAGE } from "./frontend/e2e/image.ts"; console.log(PLAYWRIGHT_IMAGE)')
echo "== browser ($image)"
podman run --rm --network host --shm-size 1g --cap-drop ALL --security-opt no-new-privileges \
  -v "$root/.e2e-1147-src:/work:ro,Z" -v "$work/pw:/pw:ro,Z" -v "$work/out:/out:Z" \
  -w /work "$image" node e2e.mjs
rc=$?
kill "$vite" 2> /dev/null
for shot in "$work"/out/*.jpg; do
  [ -f "$shot" ] && echo "SHOT $(basename "$shot") $(base64 -w0 "$shot")"
done
echo "browser=$rc"
exit "$rc"
