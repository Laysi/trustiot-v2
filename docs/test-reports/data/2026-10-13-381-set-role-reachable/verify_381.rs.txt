fn fill(sql: &str, n: usize) -> String {
    sql.replace("{T}", T)
        .replace("{A}", A)
        .replace("{B}", B)
        .replace("{p}", &format!("v{n}_p"))
        .replace("{g}", &format!("v{n}_g"))
        .replace("{h}", &format!("v{n}_h"))
}

async fn run(connection: &mut PgConnection, sql: String) {
    sqlx::raw_sql(AssertSqlSafe(sql.clone()))
        .execute(connection)
        .await
        .unwrap_or_else(|error| panic!("{sql}: {error}"));
}

fn short(error: &sqlx::Error) -> String {
    match error {
        sqlx::Error::Database(db) => {
            format!("ERR {} {}", db.code().unwrap_or_default(), db.message())
        }
        other => format!("ERR {other}"),
    }
}

/// The session check of main 9beb8f7 (Laysi/trustiot-v2#375), unchanged.
const OLD_SESSION_CHECK: &str = r"
WITH RECURSIVE rels AS (
    SELECT c.oid, c.relkind, c.relowner, c.relrowsecurity, c.reloptions, c.relnamespace,
           format('%I.%I', n.nspname, c.relname) AS name,
           EXISTS (
               SELECT 1 FROM pg_attribute a
               WHERE a.attrelid = c.oid AND a.attname = 'tenant_id'
                 AND a.attnum > 0 AND NOT a.attisdropped
           ) AS has_tenant
    FROM pg_class c
    JOIN pg_namespace n ON n.oid = c.relnamespace
    WHERE c.relkind IN ('r', 'p', 'v', 'm', 'f')
      AND n.nspname NOT IN ('pg_catalog', 'information_schema')
      AND n.nspname NOT LIKE 'pg\_toast%'
      AND n.nspname NOT LIKE 'pg\_temp%'
),
tenant_data(oid) AS (
    SELECT oid FROM rels WHERE has_tenant
    UNION
    SELECT rw.ev_class
    FROM pg_rewrite rw
    JOIN pg_depend d ON d.classid = 'pg_rewrite'::regclass AND d.objid = rw.oid
                    AND d.refclassid = 'pg_class'::regclass
    JOIN tenant_data t ON t.oid = d.refobjid
    WHERE rw.ev_class <> d.refobjid
),
reachable AS (
    SELECT * FROM rels
    WHERE oid IN (SELECT oid FROM tenant_data)
      AND has_schema_privilege(relnamespace, 'USAGE')
      AND (has_table_privilege(oid, 'SELECT, INSERT, UPDATE, DELETE, TRUNCATE')
           OR has_any_column_privilege(oid, 'SELECT, INSERT, UPDATE'))
),
-- Platform sessions only: every role the session can switch to, itself included.
-- The rights of roles it only inherits are already the session role's own; a role it can
-- neither inherit nor become gives it nothing.
platform_roles AS (
    SELECT oid, quote_ident(rolname) AS role_name FROM pg_roles
    WHERE $1 AND pg_has_role(current_user, oid, 'SET')
),
-- The tables with row-level security that one of those roles may use at all.
platform_tables AS MATERIALIZED (
    SELECT pr.oid AS role_oid, r.oid AS rel
    FROM platform_roles pr
    CROSS JOIN rels r
    WHERE r.relkind IN ('r', 'p') AND r.relrowsecurity
      AND has_schema_privilege(pr.oid, r.relnamespace, 'USAGE')
      AND (has_table_privilege(pr.oid, r.oid, 'SELECT, INSERT, UPDATE, DELETE')
           OR has_any_column_privilege(pr.oid, r.oid, 'SELECT, INSERT, UPDATE'))
),
-- The permissive policies, each printed once. Restrictive policies only narrow access, so
-- they are not counted. opaque: the policy's expressions use an object other than its own
-- table and columns, such as a function or operator that is not built in (built-in catalog
-- objects are pinned and record no dependency), another relation (a view, or a table whose own
-- policies read the tenant), or a type. The check cannot see what such an object does, so it
-- cannot tell whether the expression reads the tenant setting.
platform_policy_exprs AS MATERIALIZED (
    SELECT pol.polrelid, pol.polcmd::text AS polcmd, pol.polroles,
           pg_get_expr(pol.polqual, pol.polrelid) AS using_expr,
           pg_get_expr(coalesce(pol.polwithcheck, pol.polqual), pol.polrelid) AS check_expr,
           EXISTS (
               SELECT 1 FROM pg_depend d
               WHERE d.classid = 'pg_policy'::regclass AND d.objid = pol.oid
                 AND NOT (d.refclassid = 'pg_class'::regclass AND d.refobjid = pol.polrelid)
           ) AS opaque
    FROM pg_policy pol
    WHERE $1 AND pol.polpermissive
),
-- A command evaluates USING to filter rows (SELECT, UPDATE, DELETE) and WITH CHECK, or USING
-- when a policy has none, to check new rows (INSERT, UPDATE). The five sides as bits: 1 SELECT,
-- 2 INSERT, 4 UPDATE USING, 8 UPDATE WITH CHECK, 16 DELETE. A policy counts for its own
-- command, and a FOR ALL policy for every command. Per policy: has, the sides it gives an
-- expression; wider, the sides where that expression opens the table beyond the session's
-- tenant. These count as opening nothing beyond it: an expression that names the tenant
-- setting, calls current_setting, or is false, and every expression of an opaque policy, since
-- it may read the setting where the check cannot see it. Any other expression counts as
-- opening the table. Counting an expression as opening nothing can only refuse more sessions,
-- never accept one, so an expression that reads the setting and still opens every row
-- (`<tenant condition> OR true`) is refused too: the check does not try to tell such
-- expressions apart.
platform_policies AS MATERIALIZED (
    SELECT e.polrelid, e.polroles, 0 = ANY (e.polroles) AS public,
           CASE WHEN e.using_expr IS NOT NULL THEN s.using_sides ELSE 0 END
           | CASE WHEN e.check_expr IS NOT NULL THEN s.check_sides ELSE 0 END AS has,
           CASE WHEN e.using_expr <> $2 AND e.using_expr <> 'false'
                     AND strpos(e.using_expr, '''trustiot.tenant_id''') = 0
                     AND strpos(e.using_expr, 'current_setting(') = 0 AND NOT e.opaque
                THEN s.using_sides ELSE 0 END
           | CASE WHEN e.check_expr <> $2 AND e.check_expr <> 'false'
                       AND strpos(e.check_expr, '''trustiot.tenant_id''') = 0
                       AND strpos(e.check_expr, 'current_setting(') = 0 AND NOT e.opaque
                  THEN s.check_sides ELSE 0 END AS wider
    FROM platform_policy_exprs e
    CROSS JOIN LATERAL (
        SELECT CASE e.polcmd WHEN 'r' THEN 1 WHEN 'w' THEN 4 WHEN 'd' THEN 16 WHEN '*' THEN 21
                             ELSE 0 END AS using_sides,
               CASE e.polcmd WHEN 'a' THEN 2 WHEN 'w' THEN 8 WHEN '*' THEN 10
                             ELSE 0 END AS check_sides
    ) AS s
),
-- A policy applies to a role that has the privileges of (inherits) one of the policy's roles,
-- or to every role (PUBLIC). The sides that the PUBLIC policies give, per table: the same for
-- every role, so they are worked out once.
platform_public AS (
    SELECT polrelid, bit_or(has) AS has, bit_or(wider) AS wider
    FROM platform_policies
    WHERE public
    GROUP BY polrelid
),
-- Which platform role inherits which role a policy names, asked once per pair instead of
-- once per table, command, and policy. A PUBLIC policy already applies to every role.
platform_inherits AS MATERIALIZED (
    SELECT pr.oid AS role_oid, named.role AS policy_role
    FROM platform_roles pr
    CROSS JOIN (SELECT DISTINCT role FROM platform_policies, unnest(polroles) AS role
                WHERE NOT public) AS named
    WHERE pg_has_role(pr.oid, named.role, 'USAGE')
),
-- The sides that the policies naming a role the platform role inherits give, per role and
-- table.
platform_named AS (
    SELECT i.role_oid, p.polrelid, bit_or(p.has) AS has, bit_or(p.wider) AS wider
    FROM platform_policies p
    CROSS JOIN LATERAL unnest(p.polroles) AS named(role)
    JOIN platform_inherits i ON i.policy_role = named.role
    WHERE NOT p.public
    GROUP BY i.role_oid, p.polrelid
),
-- The commands only the tenant condition opens to the role: every side the command evaluates
-- has an expression, and on at least one side none of them opens the table beyond the tenant.
-- A role and table that no policy applies to has no expression, so nothing is tenant-only.
platform_tenant_only AS MATERIALIZED (
    SELECT t.role_oid, t.rel, c.command
    FROM platform_tables t
    LEFT JOIN platform_public pub ON pub.polrelid = t.rel
    LEFT JOIN platform_named n ON n.role_oid = t.role_oid AND n.polrelid = t.rel
    CROSS JOIN LATERAL (
        SELECT coalesce(pub.has, 0) | coalesce(n.has, 0) AS has,
               coalesce(pub.wider, 0) | coalesce(n.wider, 0) AS wider
    ) AS m
    JOIN (VALUES ('SELECT', 1), ('INSERT', 2), ('UPDATE', 12), ('DELETE', 16))
         AS c(command, sides)
      ON (m.has & c.sides) = c.sides AND (m.wider & c.sides) <> c.sides
)
SELECT problem, relation, role_name, command FROM (
    SELECT 1 AS rank, 'bypasses_rls' AS problem, NULL::text AS relation,
           NULL::text AS role_name, NULL::text AS command
    WHERE EXISTS (
        SELECT 1 FROM pg_roles r
        WHERE (r.rolsuper OR r.rolbypassrls) AND pg_has_role(current_user, r.oid, 'MEMBER')
    )
    UNION ALL
    SELECT 2, 'tenant_preset', NULL, NULL, NULL
    WHERE NULLIF(current_setting('trustiot.tenant_id', true), '') IS NOT NULL
    UNION ALL
    SELECT 3, 'owns_relation', name, NULL, NULL FROM rels
    WHERE pg_has_role(current_user, relowner, 'MEMBER')
    UNION ALL
    SELECT 4, 'unprotected_table', name, NULL, NULL FROM reachable
    WHERE relkind IN ('r', 'p') AND NOT relrowsecurity
    UNION ALL
    SELECT 4, 'permissive_policy', r.name || ' policy ' || quote_ident(p.polname), NULL, NULL
    FROM reachable r
    JOIN pg_policy p ON p.polrelid = r.oid
    WHERE r.relkind IN ('r', 'p') AND r.relrowsecurity AND p.polpermissive
      AND (0 = ANY (p.polroles)
           OR EXISTS (SELECT 1 FROM unnest(p.polroles) AS role
                      WHERE role <> 0 AND pg_has_role(current_user, role, 'MEMBER')))
      AND NOT ($1 AND NOT 0 = ANY (p.polroles)
               AND (SELECT oid FROM pg_roles WHERE rolname = current_user) = ANY (p.polroles))
      AND NOT ((p.polqual IS NOT NULL OR p.polwithcheck IS NOT NULL)
               AND (p.polqual IS NULL OR pg_get_expr(p.polqual, p.polrelid) = $2)
               AND (p.polwithcheck IS NULL OR pg_get_expr(p.polwithcheck, p.polrelid) = $2))
    UNION ALL
    SELECT 4, 'unprotected_view', name, NULL, NULL FROM reachable
    WHERE relkind = 'v'
      AND NOT EXISTS (
          SELECT 1 FROM unnest(reloptions) AS option
          WHERE lower(option) IN ('security_invoker=true', 'security_invoker=on',
                                  'security_invoker=yes', 'security_invoker=1')
      )
    UNION ALL
    SELECT 4, 'unprotectable', name, NULL, NULL FROM reachable WHERE relkind IN ('m', 'f')
    UNION ALL
    SELECT 4, 'definer_function', p.oid::regprocedure::text, NULL, NULL
    FROM pg_proc p
    JOIN pg_namespace n ON n.oid = p.pronamespace
    WHERE p.prosecdef
      AND n.nspname NOT IN ('pg_catalog', 'information_schema')
      AND NOT pg_has_role(current_user, p.proowner, 'MEMBER')
      AND has_schema_privilege(n.oid, 'USAGE')
      AND has_function_privilege(p.oid, 'EXECUTE')
      AND NOT EXISTS (
          SELECT 1 FROM pg_depend d
          WHERE d.classid = 'pg_proc'::regclass AND d.objid = p.oid AND d.deptype = 'e'
      )
    UNION ALL
    SELECT 4, 'temporary_tables', NULL, NULL, NULL
    WHERE has_database_privilege(current_database(), 'TEMPORARY')
    UNION ALL
    SELECT 5, 'truncate_allowed', name, NULL, NULL FROM reachable
    WHERE relkind IN ('r', 'p') AND has_table_privilege(oid, 'TRUNCATE')
    UNION ALL
    -- The commands only the tenant condition opens to a platform role that the role may run,
    -- through a table or column privilege. DELETE has no column privilege, and NULLIF makes its
    -- column test NULL.
    SELECT 4, 'platform_tenant_only', r.name, pr.role_name, o.command
    FROM platform_tenant_only o
    JOIN platform_roles pr ON pr.oid = o.role_oid
    JOIN rels r ON r.oid = o.rel
    WHERE has_table_privilege(o.role_oid, o.rel, o.command)
       OR coalesce(has_any_column_privilege(o.role_oid, o.rel, NULLIF(o.command, 'DELETE')), false)
) AS problems
ORDER BY rank, relation, problem, role_name, command
LIMIT 1
";

// ---------------------------------------------------------------------------------------------
// Laysi/trustiot-v2#381: the session check of main 9beb8f7 (`OLD_SESSION_CHECK` above) against
// the one in src/postgres.rs, which also judges every role the session can become with SET ROLE.
//
// - `differential`: both queries, without their `LIMIT 1`, over the same catalogue states (the
//   #366 verification cases above and seeded random states), as the platform role and as a
//   service role, plus the first row each reports (what check_session returns). Per state:
//   C1  the new rows for the session's own role (role column cleared) and the platform rule's
//       rows are exactly main's rows, in every state;
//   C2  where every role the session can SET ROLE to is also one it inherits (no SET-only
//       membership), the new rows add no (problem, relation) that main lacks: projected without
//       the role, they are exactly main's rows;
//   C3  the first row: without a SET-only membership it is main's first row (naming the session
//       role); with one, the new query refuses whenever main refuses; and when the session role
//       itself has the reported problem, the session role is the one named;
//   C4  the new rows for each role the session can become are exactly what an independent
//       per-role statement of the rule (`ORACLE`) finds for that role, and no row names a role
//       the session cannot become. A role the session also inherits is skipped by the check:
//       no row names it, and everything the oracle finds for it is among the session role's
//       own problems;
//   C5  for the problems that depend only on privileges, the new rows for each such role are
//       exactly main's rows when the superuser runs main's query as that role (SET ROLE); for a
//       skipped role, those rows are among the session role's own problems.
// - `mutations`: deliberate changes to the new query, each of which the differential must catch.
// - `timing`: both queries at 300 template tables: a service role alone and with 20 and 60
//   SET-only groups that hold every privilege on every table, and the #375 platform scenarios.
//
// Catalogue states are built inside one transaction of the superuser, which then runs the
// queries after `SET ROLE` to the session role (`current_user` is all the check looks at) and
// rolls everything back, roles included.
// ---------------------------------------------------------------------------------------------

/// The four columns of a problem row.
type Row = (String, Option<String>, Option<String>, Option<String>);

/// The problems the new query finds for each role the session can become.
const PER_ROLE: &[&str] = &[
    "unprotected_table",
    "permissive_policy",
    "unprotected_view",
    "unprotectable",
    "definer_function",
    "temporary_tables",
    "truncate_allowed",
];

/// Of those, the ones that depend only on the role's privileges.
const PRIVILEGE_ONLY: &[&str] = &[
    "unprotected_table",
    "unprotected_view",
    "unprotectable",
    "temporary_tables",
    "truncate_allowed",
];

/// The new query, as committed in src/postgres.rs.
fn new_session_check() -> String {
    let source = std::fs::read_to_string(concat!(env!("CARGO_MANIFEST_DIR"), "/src/postgres.rs"))
        .expect("source");
    let marker = "const SESSION_CHECK: &str = r\"";
    let start = source.find(marker).expect("start") + marker.len();
    let end = start + source[start..].find("\";").expect("end");
    source[start..end].to_owned()
}

/// `sql` without its `LIMIT 1`, so that every problem row comes back.
fn every_row(sql: &str) -> String {
    assert_eq!(sql.matches("\nLIMIT 1\n").count(), 1, "one LIMIT 1");
    sql.replace("\nLIMIT 1\n", "\n")
}

/// An independent statement of the #381 rule for one role (`$3`, its name) that the session can
/// become: the problems that role's privileges give the session, written for one role at a time,
/// with a policy's roles tested one by one, and without the production query's shared CTEs.
/// `$1` and `$2` as in the check.
const ORACLE: &str = r"
WITH RECURSIVE rels AS (
    SELECT c.oid, c.relkind, c.relrowsecurity, c.reloptions, c.relnamespace,
           format('%I.%I', n.nspname, c.relname) AS name,
           EXISTS (
               SELECT 1 FROM pg_attribute a
               WHERE a.attrelid = c.oid AND a.attname = 'tenant_id'
                 AND a.attnum > 0 AND NOT a.attisdropped
           ) AS has_tenant
    FROM pg_class c
    JOIN pg_namespace n ON n.oid = c.relnamespace
    WHERE c.relkind IN ('r', 'p', 'v', 'm', 'f')
      AND n.nspname NOT IN ('pg_catalog', 'information_schema')
      AND n.nspname NOT LIKE 'pg\_toast%'
      AND n.nspname NOT LIKE 'pg\_temp%'
),
tenant_data(oid) AS (
    SELECT oid FROM rels WHERE has_tenant
    UNION
    SELECT rw.ev_class
    FROM pg_rewrite rw
    JOIN pg_depend d ON d.classid = 'pg_rewrite'::regclass AND d.objid = rw.oid
                    AND d.refclassid = 'pg_class'::regclass
    JOIN tenant_data t ON t.oid = d.refobjid
    WHERE rw.ev_class <> d.refobjid
),
target AS (
    SELECT oid, quote_ident(rolname) AS role_name, rolname = current_user AS own
    FROM pg_roles WHERE rolname::text = $3
),
usable AS (
    SELECT r.* FROM rels r, target t
    WHERE r.oid IN (SELECT oid FROM tenant_data)
      AND has_schema_privilege(t.oid, r.relnamespace, 'USAGE')
      AND (has_table_privilege(t.oid, r.oid, 'SELECT')
           OR has_table_privilege(t.oid, r.oid, 'INSERT')
           OR has_table_privilege(t.oid, r.oid, 'UPDATE')
           OR has_table_privilege(t.oid, r.oid, 'DELETE')
           OR has_table_privilege(t.oid, r.oid, 'TRUNCATE')
           OR has_any_column_privilege(t.oid, r.oid, 'SELECT')
           OR has_any_column_privilege(t.oid, r.oid, 'INSERT')
           OR has_any_column_privilege(t.oid, r.oid, 'UPDATE'))
)
SELECT found.problem, found.relation, (SELECT role_name FROM target), NULL::text
FROM (
    SELECT 'unprotected_table' AS problem, name AS relation FROM usable
    WHERE relkind IN ('r', 'p') AND NOT relrowsecurity
    UNION ALL
    SELECT 'permissive_policy', u.name || ' policy ' || quote_ident(p.polname)
    FROM usable u
    JOIN pg_policy p ON p.polrelid = u.oid
    CROSS JOIN target t
    WHERE u.relkind IN ('r', 'p') AND u.relrowsecurity AND p.polpermissive
      AND (0 = ANY (p.polroles)
           OR EXISTS (SELECT 1 FROM unnest(p.polroles) AS x
                      WHERE x <> 0
                        AND (pg_has_role(t.oid, x, 'USAGE')
                             OR (t.own AND pg_has_role(current_user, x, 'MEMBER')))))
      AND NOT ($1 AND NOT 0 = ANY (p.polroles)
               AND (SELECT oid FROM pg_roles WHERE rolname = current_user) = ANY (p.polroles))
      AND NOT ((p.polqual IS NOT NULL OR p.polwithcheck IS NOT NULL)
               AND (p.polqual IS NULL OR pg_get_expr(p.polqual, p.polrelid) = $2)
               AND (p.polwithcheck IS NULL OR pg_get_expr(p.polwithcheck, p.polrelid) = $2))
    UNION ALL
    SELECT 'unprotected_view', name FROM usable
    WHERE relkind = 'v'
      AND NOT coalesce((SELECT bool_or(lower(o) IN ('security_invoker=true', 'security_invoker=on',
                                                    'security_invoker=yes', 'security_invoker=1'))
                        FROM unnest(reloptions) AS o), false)
    UNION ALL
    SELECT 'unprotectable', name FROM usable WHERE relkind IN ('m', 'f')
    UNION ALL
    SELECT 'definer_function', p.oid::regprocedure::text
    FROM pg_proc p
    JOIN pg_namespace n ON n.oid = p.pronamespace
    CROSS JOIN target t
    WHERE p.prosecdef
      AND n.nspname NOT IN ('pg_catalog', 'information_schema')
      AND NOT pg_has_role(current_user, p.proowner, 'USAGE')
      AND NOT pg_has_role(current_user, p.proowner, 'SET')
      AND has_schema_privilege(t.oid, n.oid, 'USAGE')
      AND has_function_privilege(t.oid, p.oid, 'EXECUTE')
      AND NOT EXISTS (
          SELECT 1 FROM pg_depend d
          WHERE d.classid = 'pg_proc'::regclass AND d.objid = p.oid AND d.deptype = 'e'
      )
    UNION ALL
    SELECT 'temporary_tables', NULL FROM target t
    WHERE has_database_privilege(t.oid, current_database(), 'TEMPORARY')
    UNION ALL
    SELECT 'truncate_allowed', u.name FROM usable u CROSS JOIN target t
    WHERE u.relkind IN ('r', 'p') AND has_table_privilege(t.oid, u.oid, 'TRUNCATE')
) AS found
";

async fn savepoint(connection: &mut PgConnection) {
    sqlx::raw_sql("SAVEPOINT q")
        .execute(&mut *connection)
        .await
        .expect("savepoint");
}

/// Ends the savepoint: a failed query aborted the transaction, so it rolls back to it.
async fn end_savepoint(connection: &mut PgConnection, ok: bool) {
    let undo = if ok {
        "RELEASE SAVEPOINT q"
    } else {
        "ROLLBACK TO SAVEPOINT q"
    };
    sqlx::raw_sql(undo)
        .execute(&mut *connection)
        .await
        .expect("savepoint end");
}

/// Every row of `sql`, sorted; `role` binds `$3`.
async fn rows_of(
    connection: &mut PgConnection,
    sql: &str,
    platform: bool,
    role: Option<&str>,
) -> Result<Vec<Row>, String> {
    savepoint(connection).await;
    let mut query = sqlx::query_as::<_, Row>(AssertSqlSafe(sql.to_owned()))
        .bind(platform)
        .bind(postgres::TENANT_CONDITION_AS_STORED);
    if let Some(role) = role {
        query = query.bind(role.to_owned());
    }
    let result = query
        .fetch_all(&mut *connection)
        .await
        .map_err(|error| short(&error));
    end_savepoint(connection, result.is_ok()).await;
    result.map(|mut rows| {
        rows.sort();
        rows
    })
}

/// The row `sql` (with its `LIMIT 1`) reports: what check_session returns.
async fn first_of(
    connection: &mut PgConnection,
    sql: &str,
    platform: bool,
) -> Result<Option<Row>, String> {
    savepoint(connection).await;
    let result = sqlx::query_as::<_, Row>(AssertSqlSafe(sql.to_owned()))
        .bind(platform)
        .bind(postgres::TENANT_CONDITION_AS_STORED)
        .fetch_optional(&mut *connection)
        .await
        .map_err(|error| short(&error));
    end_savepoint(connection, result.is_ok()).await;
    result
}

async fn set_role(connection: &mut PgConnection, role: Option<&str>) {
    let sql = role.map_or_else(
        || "RESET ROLE".to_owned(),
        |role| format!("SET ROLE {role}"),
    );
    sqlx::raw_sql(AssertSqlSafe(sql.clone()))
        .execute(&mut *connection)
        .await
        .unwrap_or_else(|error| panic!("{sql}: {error}"));
}

/// The four queries of a state, by `$1` (index 0: platform, 1: service).
struct Sides<T> {
    platform: T,
    service: T,
}

/// What one state gives.
struct Observed {
    old: Sides<Result<Vec<Row>, String>>,
    new: Sides<Result<Vec<Row>, String>>,
    old_first: Sides<Result<Option<Row>, String>>,
    new_first: Sides<Result<Option<Row>, String>>,
    /// The session role, quoted as the check prints it.
    session: String,
    /// The roles the session can become: (name, quoted name, whether the session also inherits
    /// it and it is not the session's own role, so that the check skips it).
    roles: Vec<(String, String, bool)>,
    /// Whether one of them is not also inherited (a SET-only membership somewhere).
    set_only: bool,
    /// SECURITY DEFINER functions main exempted (the session belongs to the owner) whose owner
    /// it neither inherits nor can become: the new query may report them for the session role.
    unjudged_definers: BTreeSet<String>,
    /// The roles the session administers, quoted.
    administered: BTreeSet<String>,
    /// Per role (as in `roles`): ORACLE's rows, platform and service.
    oracle: Vec<Sides<Result<Vec<Row>, String>>>,
    /// Per role: main's rows when run as that role (service rules).
    main_as: Vec<Result<Vec<Row>, String>>,
}

async fn observe(
    connection: &mut PgConnection,
    setup: &str,
    session: &str,
    old: &str,
    new: &str,
) -> Observed {
    let (old_all, new_all) = (every_row(old), every_row(new));
    sqlx::raw_sql("BEGIN")
        .execute(&mut *connection)
        .await
        .expect("begin");
    sqlx::raw_sql(AssertSqlSafe(setup.to_owned()))
        .execute(&mut *connection)
        .await
        .unwrap_or_else(|error| panic!("{setup}: {error}"));
    set_role(connection, Some(session)).await;
    let old_rows = Sides {
        platform: rows_of(connection, &old_all, true, None).await,
        service: rows_of(connection, &old_all, false, None).await,
    };
    let new_rows = Sides {
        platform: rows_of(connection, &new_all, true, None).await,
        service: rows_of(connection, &new_all, false, None).await,
    };
    let old_first = Sides {
        platform: first_of(connection, old, true).await,
        service: first_of(connection, old, false).await,
    };
    let new_first = Sides {
        platform: first_of(connection, new, true).await,
        service: first_of(connection, new, false).await,
    };
    let session_quoted: String = sqlx::query_scalar("SELECT quote_ident(current_user)")
        .fetch_one(&mut *connection)
        .await
        .expect("session");
    let roles: Vec<(String, String, bool)> = sqlx::query_as(
        "SELECT rolname::text, quote_ident(rolname),
                rolname <> current_user AND pg_has_role(current_user, oid, 'USAGE')
         FROM pg_roles WHERE pg_has_role(current_user, oid, 'SET') ORDER BY rolname",
    )
    .fetch_all(&mut *connection)
    .await
    .expect("roles");
    let unjudged_definers: BTreeSet<String> = sqlx::query_scalar::<_, String>(
        "SELECT p.oid::regprocedure::text FROM pg_proc p
         JOIN pg_namespace n ON n.oid = p.pronamespace
         WHERE p.prosecdef AND n.nspname NOT IN ('pg_catalog', 'information_schema')
           AND pg_has_role(current_user, p.proowner, 'MEMBER')
           AND NOT pg_has_role(current_user, p.proowner, 'USAGE')
           AND NOT pg_has_role(current_user, p.proowner, 'SET')",
    )
    .fetch_all(&mut *connection)
    .await
    .expect("unjudged definers")
    .into_iter()
    .collect();
    let administered: BTreeSet<String> = sqlx::query_scalar::<_, String>(
        "SELECT quote_ident(rolname) FROM pg_roles
         WHERE pg_has_role(current_user, oid, 'MEMBER WITH ADMIN OPTION')",
    )
    .fetch_all(&mut *connection)
    .await
    .expect("administered")
    .into_iter()
    .collect();
    let set_only: bool = sqlx::query_scalar(
        "SELECT EXISTS (SELECT 1 FROM pg_roles WHERE pg_has_role(current_user, oid, 'SET')
                                                 AND NOT pg_has_role(current_user, oid, 'USAGE'))",
    )
    .fetch_one(&mut *connection)
    .await
    .expect("set only");
    let mut oracle = Vec::new();
    for (name, _, _) in &roles {
        oracle.push(Sides {
            platform: rows_of(connection, ORACLE, true, Some(name)).await,
            service: rows_of(connection, ORACLE, false, Some(name)).await,
        });
    }
    let mut main_as = Vec::new();
    for (_, quoted, _) in &roles {
        set_role(connection, None).await;
        set_role(connection, Some(quoted)).await;
        main_as.push(rows_of(connection, &old_all, false, None).await);
    }
    set_role(connection, None).await;
    sqlx::raw_sql("ROLLBACK")
        .execute(&mut *connection)
        .await
        .expect("rollback");
    Observed {
        old: old_rows,
        new: new_rows,
        old_first,
        new_first,
        session: session_quoted,
        roles,
        set_only,
        unjudged_definers,
        administered,
        oracle,
        main_as,
    }
}

/// `row` with its role cleared when it is a per-role problem of the session's own role; `None`
/// when it is a per-role problem of another role.
fn as_main(row: &Row, session: &str) -> Option<Row> {
    if PER_ROLE.contains(&row.0.as_str()) {
        (row.2.as_deref() == Some(session)).then(|| (row.0.clone(), row.1.clone(), None, None))
    } else {
        Some(row.clone())
    }
}

fn projection(rows: &[Row]) -> BTreeSet<(String, Option<String>)> {
    rows.iter()
        .map(|row| (row.0.clone(), row.1.clone()))
        .collect()
}

/// What the differential saw over all states.
#[derive(Default)]
struct Tally {
    states: usize,
    set_only_states: usize,
    /// States where a rule beyond SET-only memberships applies (an unjudged definer owner or
    /// the ADMIN option).
    widened_states: usize,
    failures: Vec<String>,
    /// States where a query failed.
    errors: usize,
    /// SET-only states where the new query reports a (problem, relation) main does not
    /// (platform, service).
    added_states: [usize; 2],
    /// Such rows per problem, both rules together.
    added_rows: BTreeMap<String, usize>,
    /// States main accepts and the new query refuses (platform, service).
    newly_refused: [usize; 2],
    /// Rows naming a role other than the session's own (both rules).
    other_role_rows: usize,
    /// The roles the session can become per state beyond its own, summed.
    extra_roles: usize,
    /// Of those, the ones the session also inherits (skipped by the check), summed.
    skipped_roles: usize,
    /// What check_session reports with the new query as the service role.
    first_problems: BTreeMap<String, usize>,
}

impl Tally {
    fn judge(&mut self, label: &str, seen: &Observed, setup: &str) {
        self.states += 1;
        if seen.set_only {
            self.set_only_states += 1;
        }
        let widened =
            seen.set_only || !seen.unjudged_definers.is_empty() || !seen.administered.is_empty();
        if widened && !seen.set_only {
            self.widened_states += 1;
        }
        self.extra_roles += seen.roles.len().saturating_sub(1);
        let mut failures: Vec<String> = Vec::new();
        let mut fail = |what: String| {
            failures.push(format!("{label}: {what}\n  setup:\n{setup}"));
        };
        let sides = [
            (
                "platform",
                &seen.old.platform,
                &seen.new.platform,
                &seen.old_first.platform,
                &seen.new_first.platform,
            ),
            (
                "service",
                &seen.old.service,
                &seen.new.service,
                &seen.old_first.service,
                &seen.new_first.service,
            ),
        ];
        let quoted: BTreeSet<&str> = seen.roles.iter().map(|(_, q, _)| q.as_str()).collect();
        self.skipped_roles += seen.roles.iter().filter(|(_, _, skipped)| *skipped).count();
        for (index, (side, old, new, old_first, new_first)) in sides.into_iter().enumerate() {
            let (Ok(old), Ok(new), Ok(old_first), Ok(new_first)) = (old, new, old_first, new_first)
            else {
                self.errors += 1;
                fail(format!(
                    "{side}: a query failed: {old:?} {new:?} {old_first:?} {new_first:?}"
                ));
                continue;
            };
            // C1
            let mut own: Vec<Row> = new
                .iter()
                .filter_map(|row| as_main(row, &seen.session))
                .collect();
            own.sort();
            let mut rest = own.clone();
            let mut missing = Vec::new();
            for row in old {
                match rest.iter().position(|candidate| candidate == row) {
                    Some(at) => {
                        rest.remove(at);
                    }
                    None => missing.push(row.clone()),
                }
            }
            let unexplained: Vec<&Row> = rest
                .iter()
                .filter(|row| match row.0.as_str() {
                    "definer_function" => !row
                        .1
                        .as_ref()
                        .is_some_and(|function| seen.unjudged_definers.contains(function)),
                    "administers_role" => false,
                    _ => true,
                })
                .collect();
            if !missing.is_empty() || !unexplained.is_empty() {
                fail(format!(
                    "{side} C1: own rows differ from main beyond the new rules\n  missing: {missing:?}\n  unexplained: {unexplained:?}\n  main: {old:?}\n  new:  {new:?}"
                ));
            }
            // C6: the administered roles, exactly.
            let administers: BTreeSet<String> = new
                .iter()
                .filter(|row| row.0 == "administers_role")
                .filter_map(|row| row.2.clone())
                .collect();
            if administers != seen.administered {
                fail(format!(
                    "{side} C6: administers_role rows {administers:?}, expected {:?}",
                    seen.administered
                ));
            }
            // C2
            let (old_set, new_set) = (projection(old), projection(new));
            if !widened && old_set != new_set {
                fail(format!(
                    "{side} C2: no SET-only membership, but the rows differ\n  main: {old:?}\n  new:  {new:?}"
                ));
            }
            if !old_set.is_subset(&new_set) {
                fail(format!("{side}: the new rows lack some of main's"));
            }
            let added: Vec<&(String, Option<String>)> = new_set.difference(&old_set).collect();
            if !added.is_empty() {
                self.added_states[index] += 1;
                for (problem, _) in added {
                    *self.added_rows.entry(problem.clone()).or_default() += 1;
                }
            }
            for row in new {
                if PER_ROLE.contains(&row.0.as_str()) {
                    match row.2.as_deref() {
                        Some(role) if quoted.contains(role) => {
                            if role != seen.session {
                                self.other_role_rows += 1;
                            }
                        }
                        other => fail(format!(
                            "{side}: a row names {other:?}, not a role the session can become: {row:?}"
                        )),
                    }
                }
            }
            // C3
            let normalized = new_first
                .as_ref()
                .map(|row| as_main(row, &seen.session).unwrap_or_else(|| row.clone()));
            // Whenever the session role itself has the reported problem, it is the role named.
            if let Some(row) = new_first {
                let named_other = PER_ROLE.contains(&row.0.as_str())
                    && row.2.as_deref() != Some(seen.session.as_str());
                if named_other
                    && new.iter().any(|own| {
                        own.0 == row.0
                            && own.1 == row.1
                            && own.2.as_deref() == Some(seen.session.as_str())
                    })
                {
                    fail(format!(
                        "{side} C3: the first row names {:?}, but the session role has the same problem",
                        row.2
                    ));
                }
            }
            if widened {
                if old_first.is_some() && new_first.is_none() {
                    fail(format!("{side} C3: main refuses, the new query accepts"));
                }
            } else if &normalized != old_first {
                fail(format!(
                    "{side} C3: first rows differ\n  main: {old_first:?}\n  new:  {new_first:?}"
                ));
            }
            if old_first.is_none() && new_first.is_some() {
                self.newly_refused[index] += 1;
            }
            if index == 1 {
                let first = new_first
                    .as_ref()
                    .map_or("accepted".to_owned(), |row| row.0.clone());
                *self.first_problems.entry(first).or_default() += 1;
            }
            // The session role's own per-role problems, projected: a role the session also
            // inherits is skipped, and everything it would add must be among these.
            let own_set: BTreeSet<(String, Option<String>)> = new
                .iter()
                .filter(|row| {
                    PER_ROLE.contains(&row.0.as_str())
                        && row.2.as_deref() == Some(seen.session.as_str())
                })
                .map(|row| (row.0.clone(), row.1.clone()))
                .collect();
            // C4
            for ((_, role, skipped), oracle) in seen.roles.iter().zip(&seen.oracle) {
                let oracle = if index == 0 {
                    &oracle.platform
                } else {
                    &oracle.service
                };
                let Ok(expected) = oracle else {
                    fail(format!(
                        "{side} C4: the oracle failed for {role}: {oracle:?}"
                    ));
                    continue;
                };
                let mut found: Vec<Row> = new
                    .iter()
                    .filter(|row| {
                        PER_ROLE.contains(&row.0.as_str())
                            && row.2.as_deref() == Some(role.as_str())
                    })
                    .cloned()
                    .collect();
                found.sort();
                if *skipped {
                    if !found.is_empty() || !projection(expected).is_subset(&own_set) {
                        fail(format!(
                            "{side} C4: {role} is inherited, so skipped, but the oracle finds a problem the session role lacks, or a row names it\n  oracle: {expected:?}\n  new:    {found:?}"
                        ));
                    }
                } else if &found != expected {
                    fail(format!(
                        "{side} C4: rows for {role} differ from the oracle\n  oracle: {expected:?}\n  new:    {found:?}"
                    ));
                }
            }
            // C5 (privileges only, the same under both rules)
            for ((_, role, skipped), main_as) in seen.roles.iter().zip(&seen.main_as) {
                let Ok(main_as) = main_as else {
                    fail(format!("{side} C5: main as {role} failed: {main_as:?}"));
                    continue;
                };
                let mut expected: Vec<Row> = main_as
                    .iter()
                    .filter(|row| PRIVILEGE_ONLY.contains(&row.0.as_str()))
                    .map(|row| (row.0.clone(), row.1.clone(), Some(role.clone()), None))
                    .collect();
                expected.sort();
                let mut found: Vec<Row> = new
                    .iter()
                    .filter(|row| {
                        PRIVILEGE_ONLY.contains(&row.0.as_str())
                            && row.2.as_deref() == Some(role.as_str())
                    })
                    .cloned()
                    .collect();
                found.sort();
                if *skipped {
                    if !found.is_empty() || !projection(&expected).is_subset(&own_set) {
                        fail(format!(
                            "{side} C5: {role} is inherited, so skipped, but main run as it finds a problem the session role lacks, or a row names it\n  main as role: {expected:?}\n  new:          {found:?}"
                        ));
                    }
                } else if found != expected {
                    fail(format!(
                        "{side} C5: rows for {role} differ from main run as {role}\n  main as role: {expected:?}\n  new:          {found:?}"
                    ));
                }
            }
        }
        self.failures.extend(failures);
    }
}

/// A small deterministic generator (xorshift64*), so that every state can be rebuilt from its
/// seed.
struct Rng(u64);

impl Rng {
    fn new(seed: u64) -> Self {
        Self(seed.wrapping_mul(0x9E37_79B9_7F4A_7C15) | 1)
    }
    fn next(&mut self) -> u64 {
        self.0 ^= self.0 >> 12;
        self.0 ^= self.0 << 25;
        self.0 ^= self.0 >> 27;
        self.0.wrapping_mul(0x2545_F491_4F6C_DD1D)
    }
    fn below(&mut self, n: usize) -> usize {
        usize::try_from(self.next() % u64::try_from(n).expect("n")).expect("index")
    }
    fn chance(&mut self, percent: u64) -> bool {
        self.next() % 100 < percent
    }
    fn pick<'a>(&mut self, items: &[&'a str]) -> &'a str {
        items[self.below(items.len())]
    }
}

const MEMBERSHIP_OPTIONS: &[&str] = &[
    "",
    " WITH INHERIT TRUE, SET TRUE",
    " WITH INHERIT TRUE, SET FALSE",
    " WITH INHERIT FALSE, SET TRUE",
    " WITH INHERIT FALSE, SET FALSE",
];

const EXPRESSIONS: &[&str] = &[
    "tenant_id = NULLIF(current_setting('trustiot.tenant_id', true), '')::uuid",
    "tenant_id = NULLIF(current_setting('trustiot.tenant_id', true), '')::uuid",
    "true",
    "false",
    "tenant_id IS NULL",
    "current_setting('trustiot.tenant_id', true)::uuid = tenant_id",
    "length(name) > 0",
    "tenant_id = r_current_tenant()",
    "r_always_true()",
    "tenant_id = NULLIF(current_setting('trustiot.tenant_id', true), '')::uuid OR true",
    "tenant_id IN (SELECT tenant FROM r_tenant_view)",
];

/// The SQL of random state `seed`: the #375 generator (the session role `r_p`, groups `r_g<i>`,
/// an unrelated role `r_u`, tables in `public` and maybe `r_s`, grants (table, column, `PUBLIC`,
/// predefined roles), memberships with every INHERIT and SET combination and chains of them,
/// permissive and restrictive policies for every command, some of which depend on a function or
/// a view), plus what the per-role checks look at: views with and without security_invoker,
/// materialized views, TRUNCATE, the TEMPORARY privilege, and SECURITY DEFINER functions owned by
/// the migration role or a group, each granted to random roles.
fn random_state(seed: u64) -> String {
    let mut rng = Rng::new(seed);
    let mut sql = String::from(
        "CREATE ROLE r_p; CREATE ROLE r_u;
         SET ROLE v_owner;
         CREATE FUNCTION r_current_tenant() RETURNS uuid LANGUAGE sql STABLE
             AS $f$ SELECT NULLIF(current_setting('trustiot.tenant_id', true), '')::uuid $f$;
         CREATE FUNCTION r_always_true() RETURNS boolean LANGUAGE sql IMMUTABLE AS $f$ SELECT true $f$;
         CREATE VIEW r_tenant_view WITH (security_invoker = true) AS
             SELECT NULLIF(current_setting('trustiot.tenant_id', true), '')::uuid AS tenant;
         RESET ROLE;\n",
    );
    if rng.chance(20) {
        sql.push_str("ALTER ROLE r_p NOINHERIT;\n");
    }
    if rng.chance(4) {
        sql.push_str(
            "CREATE ROLE r_a; GRANT r_a TO r_p WITH ADMIN TRUE, INHERIT FALSE, SET FALSE;\n",
        );
    }
    let groups = rng.below(5);
    let mut grantees = vec!["r_p".to_owned()];
    for i in 1..=groups {
        sql.push_str(&format!("CREATE ROLE r_g{i};\n"));
        grantees.push(format!("r_g{i}"));
    }
    for i in 1..=groups {
        if rng.chance(65) {
            let option = rng.pick(MEMBERSHIP_OPTIONS);
            sql.push_str(&format!("GRANT r_g{i} TO r_p{option};\n"));
        }
        for j in (i + 1)..=groups {
            if rng.chance(25) {
                let option = rng.pick(MEMBERSHIP_OPTIONS);
                sql.push_str(&format!("GRANT r_g{j} TO r_g{i}{option};\n"));
            }
        }
    }
    for predefined in ["pg_read_all_data", "pg_write_all_data"] {
        if rng.chance(8) {
            let to = grantees[rng.below(grantees.len())].clone();
            let option = rng.pick(MEMBERSHIP_OPTIONS);
            sql.push_str(&format!("GRANT {predefined} TO {to}{option};\n"));
        }
    }
    let other_schema = rng.chance(40);
    if other_schema {
        sql.push_str("SET ROLE v_owner; CREATE SCHEMA r_s; RESET ROLE;\n");
        for grantee in grantees.iter().map(String::as_str).chain(["PUBLIC"]) {
            if rng.chance(40) {
                sql.push_str(&format!("GRANT USAGE ON SCHEMA r_s TO {grantee};\n"));
            }
        }
    }
    let mut policy_roles: Vec<String> = grantees.clone();
    policy_roles.push("r_u".to_owned());
    let tables = 1 + rng.below(3);
    for t in 1..=tables {
        let schema = if other_schema && rng.chance(50) {
            "r_s"
        } else {
            "public"
        };
        let table = format!("{schema}.r_t{t}");
        let partitioned = rng.chance(12);
        sql.push_str("SET ROLE v_owner;\n");
        if partitioned {
            sql.push_str(&format!(
                "CREATE TABLE {table} (tenant_id uuid NOT NULL, name text NOT NULL) PARTITION BY LIST (tenant_id);
                 CREATE TABLE {schema}.r_t{t}_a PARTITION OF {table} FOR VALUES IN ('{A}');\n"
            ));
        } else {
            sql.push_str(&format!(
                "CREATE TABLE {table} (tenant_id uuid NOT NULL, name text NOT NULL);\n"
            ));
        }
        if rng.chance(80) {
            sql.push_str(&format!("ALTER TABLE {table} ENABLE ROW LEVEL SECURITY;\n"));
        }
        for p in 0..rng.below(5) {
            let restrictive = if rng.chance(20) {
                " AS RESTRICTIVE"
            } else {
                ""
            };
            let command = rng.pick(&["ALL", "ALL", "SELECT", "INSERT", "UPDATE", "DELETE"]);
            let to = if rng.chance(30) {
                String::new()
            } else {
                let mut names: Vec<&str> = policy_roles
                    .iter()
                    .map(String::as_str)
                    .filter(|_| rng.chance(40))
                    .collect();
                if names.is_empty() {
                    names.push(policy_roles[rng.below(policy_roles.len())].as_str());
                }
                format!(" TO {}", names.join(", "))
            };
            let (using, check) = match command {
                "SELECT" | "DELETE" => (true, false),
                "INSERT" => (false, true),
                _ => {
                    let using = rng.chance(75);
                    (using, !using || rng.chance(60))
                }
            };
            let mut clause = String::new();
            if using {
                clause.push_str(&format!(" USING ({})", rng.pick(EXPRESSIONS)));
            }
            if check {
                clause.push_str(&format!(" WITH CHECK ({})", rng.pick(EXPRESSIONS)));
            }
            sql.push_str(&format!(
                "CREATE POLICY r_pol{p} ON {table}{restrictive} FOR {command}{to}{clause};\n"
            ));
        }
        // Views and materialized views over the table, in the same schema.
        let view = rng.chance(35).then(|| {
            let invoker = if rng.chance(50) {
                " WITH (security_invoker = true)"
            } else {
                ""
            };
            sql.push_str(&format!(
                "CREATE VIEW {schema}.r_v{t}{invoker} AS SELECT name FROM {table};\n"
            ));
            format!("{schema}.r_v{t}")
        });
        let matview = rng.chance(15).then(|| {
            sql.push_str(&format!(
                "CREATE MATERIALIZED VIEW {schema}.r_m{t} AS SELECT tenant_id FROM {table};\n"
            ));
            format!("{schema}.r_m{t}")
        });
        sql.push_str("RESET ROLE;\n");
        for grantee in grantees.iter().map(String::as_str).chain(["PUBLIC"]) {
            if rng.chance(35) {
                let privileges: Vec<&str> = ["SELECT", "INSERT", "UPDATE", "DELETE", "TRUNCATE"]
                    .into_iter()
                    .filter(|_| rng.chance(45))
                    .collect();
                if !privileges.is_empty() {
                    sql.push_str(&format!(
                        "GRANT {} ON {table} TO {grantee};\n",
                        privileges.join(", ")
                    ));
                }
            }
            if rng.chance(15) {
                let columns =
                    rng.pick(&["SELECT (name)", "INSERT (tenant_id, name)", "UPDATE (name)"]);
                sql.push_str(&format!("GRANT {columns} ON {table} TO {grantee};\n"));
            }
            for derived in [&view, &matview].into_iter().flatten() {
                if rng.chance(30) {
                    sql.push_str(&format!("GRANT SELECT ON {derived} TO {grantee};\n"));
                }
            }
        }
    }
    // The TEMPORARY privilege (sql/database.sql revoked it from PUBLIC).
    for grantee in grantees.iter().map(String::as_str).chain(["PUBLIC"]) {
        if rng.chance(10) {
            sql.push_str(&format!(
                "DO $d$ BEGIN EXECUTE format('GRANT TEMPORARY ON DATABASE %I TO {grantee}', current_database()); END $d$;\n"
            ));
        }
    }
    // A SECURITY DEFINER function, owned by the migration role or a group.
    if rng.chance(30) {
        sql.push_str(
            "SET ROLE v_owner;
             CREATE FUNCTION r_definer() RETURNS integer LANGUAGE sql SECURITY DEFINER AS $f$ SELECT 1 $f$;
             RESET ROLE;\n",
        );
        if groups > 0 && rng.chance(30) {
            let owner = 1 + rng.below(groups);
            sql.push_str(&format!(
                "ALTER FUNCTION r_definer() OWNER TO r_g{owner};\n"
            ));
        }
        if rng.chance(70) {
            sql.push_str("REVOKE EXECUTE ON FUNCTION r_definer() FROM PUBLIC;\n");
        }
        for grantee in &grantees {
            if rng.chance(30) {
                sql.push_str(&format!(
                    "GRANT EXECUTE ON FUNCTION r_definer() TO {grantee};\n"
                ));
            }
        }
    }
    sql
}

async fn differential_over(
    connection: &mut PgConnection,
    old: &str,
    new: &str,
    seeds: u64,
    stop_at_first: bool,
) -> Tally {
    let mut tally = Tally::default();
    // The #366 verification cases (sections 1 to 4), roles v<n>_p, v<n>_g, v<n>_h.
    for (index, case) in cases().into_iter().enumerate() {
        let n = index + 1;
        let mut setup = format!(
            "CREATE ROLE v{n}_p; CREATE ROLE v{n}_g; CREATE ROLE v{n}_h;
             SET ROLE v_owner;\n{}\nRESET ROLE;\n",
            fill(BASE, n)
        );
        for step in case.setup {
            setup.push_str(&fill(step, n));
            setup.push('\n');
        }
        let seen = observe(connection, &setup, &format!("v{n}_p"), old, new).await;
        tally.judge(&format!("case {}", case.id), &seen, &setup);
        if stop_at_first && !tally.failures.is_empty() {
            return tally;
        }
    }
    for seed in 1..=seeds {
        let setup = random_state(seed);
        let seen = observe(connection, &setup, "r_p", old, new).await;
        tally.judge(&format!("seed {seed}"), &seen, &setup);
        if stop_at_first && !tally.failures.is_empty() {
            return tally;
        }
    }
    tally
}

async fn probe_database(pg: &TlsPostgres, name: &str) -> PgConnection {
    let mut admin = PgConnection::connect_with(&pg.superuser("postgres"))
        .await
        .expect("superuser");
    run(&mut admin, format!("CREATE DATABASE {name} OWNER v_owner")).await;
    admin.close().await.expect("close");
    let mut owner = PgConnection::connect_with(&pg.options("v_owner", name))
        .await
        .expect("owner");
    run(&mut owner, include_str!("../sql/database.sql").to_owned()).await;
    owner.close().await.expect("close");
    PgConnection::connect_with(&pg.superuser(name))
        .await
        .expect("superuser on the probe database")
}

const SEEDS: u64 = 1500;

#[tokio::test]
async fn differential() {
    let Some(pg) = tls_postgres(INIT).await else {
        return;
    };
    let mut connection = probe_database(&pg, "vdiff").await;
    let new = new_session_check();
    assert_ne!(
        OLD_SESSION_CHECK, new,
        "the new query differs from the old one"
    );
    let tally = differential_over(&mut connection, OLD_SESSION_CHECK, &new, SEEDS, false).await;
    println!(
        "differential: {} states ({} #366 cases + {SEEDS} seeds); {} with a SET-only membership; \
         {} roles the session can become beyond its own ({} of them also inherited, so skipped); \
         {} states where a query failed",
        tally.states,
        cases().len(),
        tally.set_only_states,
        tally.extra_roles,
        tally.skipped_roles,
        tally.errors
    );
    println!(
        "SET-only states where the new query reports a (problem, relation) main does not: \
         platform {}, service {}; such rows by problem: {:?}",
        tally.added_states[0], tally.added_states[1], tally.added_rows
    );
    println!(
        "states main accepts and the new query refuses: platform {}, service {}; rows naming a \
         role other than the session's: {}",
        tally.newly_refused[0], tally.newly_refused[1], tally.other_role_rows
    );
    println!(
        "first problem as the service role (what check_session reports with the new query): {:?}",
        tally.first_problems
    );
    println!(
        "states without a SET-only membership where an unjudged definer owner or the ADMIN \
         option applies: {}",
        tally.widened_states
    );
    println!("failures: {}", tally.failures.len());
    for failure in tally.failures.iter().take(5) {
        println!("{failure}");
    }
    assert!(tally.failures.is_empty(), "the properties do not hold");
    assert_eq!(tally.errors, 0, "every query ran");
    assert!(
        tally.added_states[1] > 100,
        "the states exercise the new rule"
    );
    assert!(
        tally.set_only_states < tally.states,
        "some states have no SET-only membership"
    );
}

/// Changes to the new query that alter its decisions; the differential must catch each.
const MUTATIONS: &[(&str, &str, &str)] = &[
    (
        "reachable: the session's own role only",
        "    WHERE r.oid IN (SELECT oid FROM tenant_data)\n      AND has_schema_privilege(sr.oid,",
        "    WHERE sr.own AND r.oid IN (SELECT oid FROM tenant_data)\n      AND has_schema_privilege(sr.oid,",
    ),
    (
        "policy applies: the session instead of the role",
        "WHERE pg_has_role(sr.oid, named.role, 'USAGE')",

#[tokio::test]
async fn mutations() {
    let Some(pg) = tls_postgres(INIT).await else {
        return;
    };
    // Its own database: the differential test may run at the same time.
    let mut connection = probe_database(&pg, "vmut").await;
    let new = new_session_check();
    let mut missed = Vec::new();
    for (name, from, to) in MUTATIONS {
        assert_eq!(
            new.matches(from).count(),
            1,
            "mutation `{name}`: `{from}` occurs once"
        );
        let mutant = new.replace(from, to);
        let tally =
            differential_over(&mut connection, OLD_SESSION_CHECK, &mutant, SEEDS, true).await;
        match tally.failures.first() {
            Some(first) => println!(
                "mutation `{name}`: caught after {} states, first at {}",
                tally.states,
                first.lines().next().unwrap_or_default()
            ),
            None => {
                println!("mutation `{name}`: NOT caught in {} states", tally.states);
                missed.push(*name);
            }
        }
    }
    assert!(
        missed.is_empty(),
        "mutations the differential missed: {missed:?}"
    );
}

// ---------------------------------------------------------------------------------------------
// Timing: 300 template tables (as in #375). The service role vc_app alone, then with 20 and 60
// groups it can become only with SET ROLE (INHERIT FALSE, SET TRUE) that hold every privilege on
// every table; and the #375 platform scenarios (the platform role vc_p alone, 20 groups without
// rights, 20 and 60 groups with every right, named in every platform_access policy).
// ---------------------------------------------------------------------------------------------

const RUNS: usize = 25;

fn loadavg() -> String {
    std::fs::read_to_string("/proc/loadavg").map_or_else(
        |error| format!("unreadable: {error}"),
        |text| text.trim().to_owned(),
    )
}

fn summary(times: &mut [Duration]) -> String {
    times.sort();
    let ms = |d: Duration| d.as_secs_f64() * 1e3;
    format!(
        "median {:.1} ms, min {:.1} ms, max {:.1} ms ({} runs)",
        ms(times[times.len() / 2]),
        ms(times[0]),
        ms(times[times.len() - 1]),
        times.len()
    )
}

async fn timed(connection: &mut PgConnection, sql: &str, platform: bool) -> (Duration, bool) {
    let started = Instant::now();
    let row: Option<Row> = sqlx::query_as(AssertSqlSafe(sql.to_owned()))
        .bind(platform)
        .bind(postgres::TENANT_CONDITION_AS_STORED)
        .fetch_optional(connection)
        .await
        .expect("query");
    (started.elapsed(), row.is_none())
}

async fn time_scenario(
    pg: &TlsPostgres,
    label: &str,
    login: &str,
    platform: bool,
    old: &str,
    new: &str,
) {
    println!("\n[{label}] loadavg before: {}", loadavg());
    let rules = if platform {
        SessionRole::Platform
    } else {
        SessionRole::Service
    };
    let mut session = PgConnection::connect_with(&pg.options(login, "vcost"))
        .await
        .expect("connect");
    let mut every = Vec::new();
    for sql in [every_row(old), every_row(new)] {
        let rows: Vec<Row> = sqlx::query_as(AssertSqlSafe(sql))
            .bind(platform)
            .bind(postgres::TENANT_CONDITION_AS_STORED)
            .fetch_all(&mut session)
            .await
            .expect("every row");
        every.push(rows);
    }
    let new_rows = every.pop().expect("new");
    let old_rows = every.pop().expect("old");
    assert!(old_rows.is_empty(), "{label}: main accepts: {old_rows:?}");
    assert!(new_rows.is_empty(), "{label}: new accepts: {new_rows:?}");
    let checked = postgres::check_session(&mut session, rules)
        .await
        .expect("check");
    let roles: i64 = sqlx::query_scalar(
        "SELECT count(*) FROM pg_roles WHERE pg_has_role(current_user, oid, 'SET')",
    )
    .fetch_one(&mut session)
    .await
    .expect("roles");
    println!(
        "[{label}] roles the session can become (itself included): {roles}; check_session: {checked:?}"
    );
    // Warm: one session, each query prepared and run once untimed, then interleaved.
    for sql in [old, new] {
        timed(&mut session, sql, platform).await;
    }
    let (mut old_warm, mut new_warm) = (Vec::new(), Vec::new());
    for _ in 0..RUNS {
        for (sql, times) in [(old, &mut old_warm), (new, &mut new_warm)] {
            let (elapsed, accepted) = timed(&mut session, sql, platform).await;
            assert!(accepted, "{label}: accepted");
            times.push(elapsed);
        }
    }
    session.close().await.expect("close");
    // Cold: the first query on a new connection, as the pool runs it.
    let (mut old_cold, mut new_cold) = (Vec::new(), Vec::new());
    for _ in 0..RUNS {
        for (sql, times) in [(old, &mut old_cold), (new, &mut new_cold)] {
            let mut fresh = PgConnection::connect_with(&pg.options(login, "vcost"))
                .await
                .expect("connect");
            let (elapsed, accepted) = timed(&mut fresh, sql, platform).await;
            assert!(accepted, "{label}: accepted");
            times.push(elapsed);
            fresh.close().await.expect("close");
        }
    }
    println!("[{label}] old, same session:   {}", summary(&mut old_warm));
    println!("[{label}] new, same session:   {}", summary(&mut new_warm));
    println!("[{label}] old, new connection: {}", summary(&mut old_cold));
    println!("[{label}] new, new connection: {}", summary(&mut new_cold));
    println!("[{label}] loadavg after: {}", loadavg());
}

#[tokio::test]
async fn timing() {
    let Some(pg) = tls_postgres(INIT).await else {
        return;
    };
    let mut admin = PgConnection::connect_with(&pg.superuser("postgres"))
        .await
        .expect("superuser");
    let version: String = sqlx::query_scalar("SELECT version()")
        .fetch_one(&mut admin)
        .await
        .expect("version");
    println!("server: {version}");
    run(
        &mut admin,
        "CREATE ROLE vc_p LOGIN PASSWORD 'service-kit-test';
         CREATE ROLE vc_app LOGIN PASSWORD 'service-kit-test';
         DO $$ BEGIN FOR i IN 1..60 LOOP
             EXECUTE format('CREATE ROLE vc_g%s NOLOGIN', i);
             EXECUTE format('CREATE ROLE vc_sg%s NOLOGIN', i);
         END LOOP; END $$;"
            .to_owned(),
    )
    .await;
    run(&mut admin, "CREATE DATABASE vcost OWNER v_owner".to_owned()).await;
    let mut owner = PgConnection::connect_with(&pg.options("v_owner", "vcost"))
        .await
        .expect("owner");
    run(&mut owner, include_str!("../sql/database.sql").to_owned()).await;
    run(
        &mut owner,
        "DO $$
         DECLARE i int;
         BEGIN
           FOR i IN 1..300 LOOP
             EXECUTE format('CREATE TABLE t%s (tenant_id uuid NOT NULL, id uuid NOT NULL, name text NOT NULL, PRIMARY KEY (tenant_id, id))', i);
             EXECUTE format('ALTER TABLE t%s ENABLE ROW LEVEL SECURITY', i);
             EXECUTE format($f$CREATE POLICY tenant_isolation ON t%s USING (tenant_id = NULLIF(current_setting('trustiot.tenant_id', true), '')::uuid) WITH CHECK (tenant_id = NULLIF(current_setting('trustiot.tenant_id', true), '')::uuid)$f$, i);
             EXECUTE format('CREATE POLICY platform_access ON t%s TO vc_p USING (true) WITH CHECK (true)', i);
             EXECUTE format('GRANT SELECT, INSERT, UPDATE, DELETE ON t%s TO vc_app', i);
             EXECUTE format('GRANT SELECT, UPDATE ON t%s TO vc_p', i);
           END LOOP;
         END $$;"
            .to_owned(),
    )
    .await;
    owner.close().await.expect("close");
    let mut db_admin = PgConnection::connect_with(&pg.superuser("vcost"))
        .await
        .expect("superuser on vcost");
    let old = OLD_SESSION_CHECK.to_owned();
    let new = new_session_check();

    time_scenario(
        &pg,
        "T0 300 tables, service role alone",
        "vc_app",
        false,
        &old,
        &new,
    )
    .await;
    run(&mut db_admin, service_groups_sql(1, 20)).await;
    time_scenario(
        &pg,
        "T1 + 20 SET-only groups with all rights on 300 tables",
        "vc_app",
        false,
        &old,
        &new,
    )
    .await;
    run(&mut db_admin, service_groups_sql(21, 60)).await;
    time_scenario(
        &pg,
        "T2 + 60 SET-only groups with all rights on 300 tables",
        "vc_app",
        false,
        &old,
        &new,
    )
    .await;

    time_scenario(
        &pg,
        "S0 300 tables, platform role alone",
        "vc_p",
        true,
        &old,
        &new,
    )
    .await;
    run(&mut db_admin, groups_sql(1, 20)).await;
    time_scenario(
        &pg,
        "S1 + 20 groups without rights",
        "vc_p",
        true,
        &old,
        &new,
    )
    .await;
    run(&mut db_admin, group_rights_sql(20)).await;
    time_scenario(
        &pg,
        "S2 20 groups with all rights on 300 tables, named in platform_access",
        "vc_p",
        true,
        &old,
        &new,
    )
    .await;
    run(&mut db_admin, groups_sql(21, 60)).await;
    run(&mut db_admin, group_rights_sql(60)).await;
    time_scenario(
        &pg,
        "S3 60 groups with all rights on 300 tables, named in platform_access",
        "vc_p",
        true,
        &old,
        &new,
    )
    .await;

    // The plan of the new query for the service role with 60 SET-only groups.
    let mut session = PgConnection::connect_with(&pg.options("vc_app", "vcost"))
        .await
        .expect("connect");
    let plan: Vec<String> =
        sqlx::query_scalar(AssertSqlSafe(format!("EXPLAIN (ANALYZE, COSTS OFF) {new}")))
            .bind(false)
            .bind(postgres::TENANT_CONDITION_AS_STORED)
            .fetch_all(&mut session)
            .await
            .expect("plan");
    for line in plan {
        println!("PLAN {line}");
    }
}

/// Groups `vc_sg<from..=to>` that the service role can become only with SET ROLE, each with
/// every privilege on every table.
fn service_groups_sql(from: usize, to: usize) -> String {
    format!(
        "DO $$
         DECLARE i int; roles text;
         BEGIN
           SELECT string_agg(format('vc_sg%s', g), ', ') INTO roles FROM generate_series({from}, {to}) g;
           EXECUTE format('GRANT %s TO vc_app WITH INHERIT FALSE, SET TRUE', roles);
           FOR i IN 1..300 LOOP
             EXECUTE format('GRANT SELECT, INSERT, UPDATE, DELETE ON t%s TO %s', i, roles);
           END LOOP;
         END $$;"
    )
}

fn groups_sql(from: usize, to: usize) -> String {
    format!(
        "DO $$
         DECLARE roles text;
         BEGIN
           SELECT string_agg(format('vc_g%s', g), ', ') INTO roles FROM generate_series({from}, {to}) g;
           EXECUTE format('GRANT %s TO vc_p', roles);
         END $$;"
    )
}

fn group_rights_sql(to: usize) -> String {
    format!(
        "DO $$
         DECLARE i int; roles text;
         BEGIN
           SELECT string_agg(format('vc_g%s', g), ', ') INTO roles FROM generate_series(1, {to}) g;
           FOR i IN 1..300 LOOP
             EXECUTE format('GRANT SELECT, INSERT, UPDATE, DELETE ON t%s TO %s', i, roles);
             EXECUTE format('ALTER POLICY platform_access ON t%s TO vc_p, %s', i, roles);
           END LOOP;
         END $$;"
    )
}
