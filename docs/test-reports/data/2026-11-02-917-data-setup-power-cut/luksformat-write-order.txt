# cryptsetup 2.7.5 luksFormat on a device: what it writes where, and in what order
# Run in a throwaway container: debian:trixie@sha256:2dc419c6d7fe3dad0a2f9fd469c34cc90b0f9ba642d8fde837fe3d7d34a0e7f1,
# cryptsetup-bin 2:2.7.5-2 from snapshot.debian.org 20261030T000000Z (the gateway image's version).

## debug.sh
#!/bin/bash
# Experiment: what cryptsetup 2.7.5 says it writes where, during luksFormat on a device (--debug).
set -euo pipefail
rm -f /etc/apt/sources.list.d/debian.sources
cat >/etc/apt/sources.list.d/snap.sources <<S
Types: deb
URIs: http://snapshot.debian.org/archive/debian/20261030T000000Z
Suites: trixie
Components: main
Signed-By: /usr/share/keyrings/debian-archive-keyring.gpg
S
apt-get -o Acquire::Check-Valid-Until=false -qq update >/dev/null
DEBIAN_FRONTEND=noninteractive apt-get -qq install -y --no-install-recommends cryptsetup-bin=2:2.7.5-2 >/dev/null
cd /tmp
head -c 48 /dev/urandom | base64 -w0 > key
truncate -s 2G direct.img
cryptsetup luksFormat --debug --batch-mode --type luks2 --cipher aes-xts-plain64 --key-size 512 --sector-size 4096 \
	--hash sha256 --pbkdf pbkdf2 --pbkdf-force-iterations 1000 --label trustiot-data-new --key-slot 0 --key-file key direct.img 2>&1 |
	grep -E 'Wiping|Writing LUKS2 header|header of size|keyslot area' | sed -E 's/^# //'

## trace.sh
#!/bin/bash
# Experiment: in what order cryptsetup 2.7.5 luksFormat writes the regions of the device.
set -euo pipefail
rm -f /etc/apt/sources.list.d/debian.sources
cat >/etc/apt/sources.list.d/snap.sources <<S
Types: deb
URIs: http://snapshot.debian.org/archive/debian/20261030T000000Z
Suites: trixie
Components: main
Signed-By: /usr/share/keyrings/debian-archive-keyring.gpg
S
apt-get -o Acquire::Check-Valid-Until=false -qq update >/dev/null
DEBIAN_FRONTEND=noninteractive apt-get -qq install -y --no-install-recommends cryptsetup-bin=2:2.7.5-2 strace >/dev/null
cd /tmp
head -c 48 /dev/urandom | base64 -w0 > key
truncate -s 2G direct.img
strace -f -s 0 -e trace=openat,write,pwrite64,pwritev,pwritev2,lseek,fsync,fdatasync -o direct.trace \
	cryptsetup luksFormat --batch-mode --type luks2 --cipher aes-xts-plain64 --key-size 512 --sector-size 4096 \
	--hash sha256 --pbkdf pbkdf2 --pbkdf-force-iterations 1000 --label trustiot-data-new --key-slot 0 --key-file key direct.img
# Only the calls on file descriptors that name direct.img.
grep -n 'direct.img' direct.trace | head -20
echo "-- writes (pid, call, fd, length/offset), consecutive duplicates counted"
grep -E 'pwrite64|pwritev|write\(|fsync|lseek' direct.trace | grep -vE 'write\((1|2),' |
	sed -E "s/^([0-9]+) +//; s/\"\"(\\.\\.\\.)?/_/" | awk "/write\\(/ { on = 1 } on" | uniq -c
