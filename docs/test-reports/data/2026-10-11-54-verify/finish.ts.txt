// Continuation after the orchestrator's drain window expired: restart consumers, wait for lag 0, verify.
import { $ } from "bun";
import { rmSync, writeFileSync, readdirSync, readFileSync, appendFileSync } from "node:fs";
const [bin, outdir] = process.argv.slice(2);
const PG = "outbox-chaos-pg", PORT = 55432;
const env = { ...process.env,
  CHAOS_URL_ADMIN: `postgres://postgres:pw@127.0.0.1:${PORT}/postgres`,
  CHAOS_URL_PRODUCER: `postgres://producer_role:producer_role@127.0.0.1:${PORT}/postgres`,
  CHAOS_URL_CONSUMER: `postgres://consumer_role:consumer_role@127.0.0.1:${PORT}/postgres` };
const sleep = (ms: number) => new Promise((r) => setTimeout(r, ms));
const psql = async (q: string) => (await $`podman exec ${PG} psql -h 127.0.0.1 -U postgres -d postgres -At -c ${q}`.env({ PGPASSWORD: "pw" }).quiet().nothrow()).stdout.toString().trim();
const log = (s: string) => { const l = `[finish ${new Date().toISOString()}] ${s}`; console.log(l); appendFileSync(`${outdir}/timeline.txt`, l + "\n"); };
rmSync(`${outdir}/STOP_CONSUMERS`, { force: true });
const cons = [["c1","run"],["c2","poll"],["c3","poll"]].map(([n, m]) => {
  const file = `${outdir}/procs/${n}-finish.log`;
  return Bun.spawn([bin, "consumer", n, m, outdir], { env, stdout: Bun.file(file), stderr: Bun.file(file + ".err") });
});
log("consumers restarted for the drain, no more faults");
const t = Date.now();
let lag = "?";
while (Date.now() - t < 40 * 60_000) {
  lag = await psql(`select count(*) from prod.outbox_event e, cons.outbox_subscription s where (e.tx_order, e.id) > (s.position_tx, s.position_id)`);
  appendFileSync(`${outdir}/lag.csv`, `finish+${((Date.now()-t)/1000).toFixed(0)},${lag},0,\n`);
  if (lag === "0") break;
  await sleep(5000);
}
log(`lag=${lag} after ${((Date.now() - t) / 1000).toFixed(0)}s of drain`);
await sleep(4000);
writeFileSync(`${outdir}/STOP_CONSUMERS`, "");
for (const c of cons) await c.exited;
const v = await $`${bin} verify ${outdir}`.env(env).quiet().nothrow();
writeFileSync(`${outdir}/verify.json`, v.stdout.toString() + v.stderr.toString());
console.log(v.stdout.toString());
const tot: Record<string, number> = { delivered: 0, dup: 0, dead: 0, handled: 0, retry_before: 0, retry_after: 0 };
for (const f of readdirSync(`${outdir}/procs`)) {
  if (!f.startsWith("c") || f.endsWith(".err")) continue;
  const last = readFileSync(`${outdir}/procs/${f}`, "utf8").split("\n").filter((l) => l.startsWith("STAT")).at(-1);
  if (!last) continue;
  for (const k of Object.keys(tot)) { const m = last.match(new RegExp(`${k}=(\\d+)`)); if (m) tot[k] += Number(m[1]); }
}
console.log(JSON.stringify({ tot }, null, 2));
writeFileSync(`${outdir}/finish-stats.json`, JSON.stringify({ tot, drain_lag_final: lag }, null, 2));
await $`podman logs ${PG}`.quiet().nothrow().then((r) => writeFileSync(`${outdir}/postgres.log`, r.stdout.toString() + r.stderr.toString()));
if (v.exitCode === 0) await $`podman rm -f ${PG}`.quiet().nothrow();
process.exit(v.exitCode === 0 ? 0 : 1);
