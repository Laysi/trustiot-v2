// Orchestrator for the outbox chaos run (verification of Laysi/trustiot-v2#54).
// Usage: bun chaos.ts <chaos-binary> <outdir> <minutes> [seed]
import { $ } from "bun";
import { mkdirSync, writeFileSync, appendFileSync, existsSync, readFileSync, readdirSync, rmSync } from "node:fs";

const [bin, outdir, minutesArg] = process.argv.slice(2);
const minutes = Number(minutesArg ?? "20");
const PG = "outbox-chaos-pg";
const PORT = 55432;
const IMAGE =
  "docker.io/library/postgres@sha256:77f585114c32fbca283dc835b0596f4e52b51b4c6662d7810b2f4084f60a1873";
const env = {
  ...process.env,
  CHAOS_URL_ADMIN: `postgres://postgres:pw@127.0.0.1:${PORT}/postgres`,
  CHAOS_URL_PRODUCER: `postgres://producer_role:producer_role@127.0.0.1:${PORT}/postgres`,
  CHAOS_URL_CONSUMER: `postgres://consumer_role:consumer_role@127.0.0.1:${PORT}/postgres`,
};
mkdirSync(`${outdir}/procs`, { recursive: true });
for (const f of readdirSync(outdir)) if (f.startsWith("producer-") || f === "STOP" || f === "STOP_CONSUMERS") rmSync(`${outdir}/${f}`);

const t0 = Date.now();
const now = () => ((Date.now() - t0) / 1000).toFixed(1);
const events: string[] = [];
const ev = (s: string) => {
  const line = `[${now()}s] ${s}`;
  events.push(line);
  appendFileSync(`${outdir}/timeline.txt`, line + "\n");
  console.log(line);
};
const sleep = (ms: number) => new Promise((r) => setTimeout(r, ms));
const rnd = (a: number, b: number) => a + Math.random() * (b - a);

const counts = {
  producer_kills: 0, consumer_kills: 0, producer_restarts: 0, consumer_restarts: 0,
  pg_graceful_restarts: 0, pg_crashes_sigkill: 0, pg_terminate_backend_calls: 0, backends_terminated: 0,
  long_write_txs_other_schema: 0, prepared_tx_across_crash: 0, replays_reprocess: 0, replays_skip: 0,
};

async function psql(sql: string, user = "postgres") {
  const r = await $`podman exec ${PG} psql -h 127.0.0.1 -U ${user} -d postgres -At -c ${sql}`.env({ PGPASSWORD: user === "postgres" ? "pw" : user }).quiet().nothrow();
  return { ok: r.exitCode === 0, out: r.stdout.toString().trim(), err: r.stderr.toString().trim() };
}
async function waitPg(timeoutS = 120) {
  const end = Date.now() + timeoutS * 1000;
  while (Date.now() < end) {
    const r = await psql("select 1");
    if (r.ok) return;
    await sleep(500);
  }
  throw new Error("postgres did not come back");
}

// ---- processes
type Proc = { name: string; kind: "producer" | "consumer"; args: string[]; p?: ReturnType<typeof Bun.spawn>; n: number; dead: boolean };
const procs: Proc[] = [
  { name: "p1", kind: "producer", args: ["producer", "p1", outdir], n: 0, dead: true },
  { name: "p2", kind: "producer", args: ["producer", "p2", outdir], n: 0, dead: true },
  { name: "p3", kind: "producer", args: ["producer", "p3", outdir], n: 0, dead: true },
  { name: "c1", kind: "consumer", args: ["consumer", "c1", "run", outdir], n: 0, dead: true },
  { name: "c2", kind: "consumer", args: ["consumer", "c2", "poll", outdir], n: 0, dead: true },
  { name: "c3", kind: "consumer", args: ["consumer", "c3", "poll", outdir], n: 0, dead: true },
];
function start(pr: Proc) {
  pr.n++;
  const file = `${outdir}/procs/${pr.name}-${String(pr.n).padStart(3, "0")}.log`;
  pr.p = Bun.spawn([bin, ...pr.args], { env, stdout: Bun.file(file), stderr: Bun.file(file + ".err") });
  pr.dead = false;
  pr.p.exited.then(() => { pr.dead = true; });
}
let stopping = false;

// ---- setup
await $`podman rm -f ${PG}`.quiet().nothrow();
ev("starting postgres container (memory 1g)");
await $`podman run -d --name ${PG} --memory 1g -e POSTGRES_PASSWORD=pw -p 127.0.0.1:${PORT}:5432 ${IMAGE} postgres -c max_connections=200 -c max_prepared_transactions=10 -c log_connections=off`.quiet();
await waitPg();
await sleep(1500);
const ver = await psql("select version()");
ev(`postgres up: ${ver.out}`);
{
  const r = await $`${bin} setup`.env(env).quiet().nothrow();
  if (r.exitCode !== 0) throw new Error("setup failed: " + r.stderr.toString() + r.stdout.toString());
  ev("setup done");
}

for (const pr of procs) start(pr);
ev("started 3 producers and 3 consumers");

const durMs = minutes * 60_000;
const elapsed = () => Date.now() - t0;

// ---- kill loop
async function killLoop() {
  while (!stopping) {
    await sleep(rnd(2500, 9000));
    if (stopping) break;
    const alive = procs.filter((p) => !p.dead);
    if (!alive.length) continue;
    const victims = [alive[Math.floor(Math.random() * alive.length)]];
    if (Math.random() < 0.15) victims.push(alive[Math.floor(Math.random() * alive.length)]);
    for (const v of new Set(victims)) {
      if (v.dead) continue;
      v.p!.kill("SIGKILL");
      if (v.kind === "producer") counts.producer_kills++; else counts.consumer_kills++;
      ev(`SIGKILL ${v.name}#${v.n}`);
      (async () => {
        await sleep(rnd(300, 4000));
        if (stopping) return;
        start(v);
        if (v.kind === "producer") counts.producer_restarts++; else counts.consumer_restarts++;
        ev(`restart ${v.name}#${v.n}`);
      })();
    }
  }
}

// ---- postgres faults
let pgDown = false;
async function pgGraceful() {
  ev("postgres: graceful restart");
  pgDown = true;
  await $`podman restart -t 10 ${PG}`.quiet().nothrow();
  await waitPg();
  pgDown = false;
  counts.pg_graceful_restarts++;
  ev("postgres: back");
}
async function pgCrash() {
  ev("postgres: SIGKILL the container (crash recovery)");
  pgDown = true;
  await $`podman kill ${PG}`.quiet().nothrow();
  await sleep(rnd(500, 3000));
  await $`podman start ${PG}`.quiet().nothrow();
  await waitPg();
  pgDown = false;
  counts.pg_crashes_sigkill++;
  ev("postgres: back after crash");
}
async function terminateLoop() {
  while (!stopping) {
    await sleep(rnd(20_000, 45_000));
    if (stopping || pgDown) continue;
    const frac = Math.random() < 0.5 ? "true" : "random() < 0.5";
    const r = await psql(`select count(*) from (select pg_terminate_backend(pid) from pg_stat_activity where usename in ('producer_role','consumer_role') and ${frac}) t`);
    counts.pg_terminate_backend_calls++;
    const n = Number(r.out) || 0;
    counts.backends_terminated += n;
    ev(`pg_terminate_backend on ${n} backends (${frac})`);
  }
}
async function longTxLoop() {
  while (!stopping) {
    await sleep(rnd(40_000, 75_000));
    if (stopping || pgDown) continue;
    const secs = Math.round(rnd(15, 35));
    counts.long_write_txs_other_schema++;
    ev(`long write transaction in schema other: ${secs}s`);
    psql(`begin; insert into other.junk default values; select pg_sleep(${secs}); commit;`).then((r) =>
      ev(`long transaction ended ok=${r.ok} ${r.err.slice(0, 80)}`));
  }
}
async function preparedAcrossCrash() {
  ev("prepared transaction in schema other, then crash of postgres while it is held");
  const r = await psql(`begin; insert into other.junk default values; prepare transaction 'chaos1';`);
  if (!r.ok) { ev("prepare failed: " + r.err); return; }
  await sleep(8000);
  await pgCrash();
  const after = await psql("select gid from pg_prepared_xacts");
  ev(`prepared transactions after crash recovery: ${after.out}`);
  await sleep(15000);
  const c = await psql("commit prepared 'chaos1'");
  ev(`commit prepared ok=${c.ok} ${c.err}`);
  counts.prepared_tx_across_crash++;
}

// ---- lag sampling (independent of the library: counts events after the subscription position)
writeFileSync(`${outdir}/lag.csv`, "t_s,lag_events,oldest_other_xact_age_s,events_total\n");
let maxLag = 0, maxLagSeen = 0;
async function lagLoop() {
  while (!stopping || true) {
    if (!pgDown) {
      const r = await psql(
        `select (select count(*) from prod.outbox_event e, cons.outbox_subscription s where (e.tx_order, e.id) > (s.position_tx, s.position_id)),
                coalesce((select extract(epoch from now() - min(xact_start))::int from pg_stat_activity where backend_xid is not null and datname = 'postgres' and usename = 'postgres' and state <> 'idle' and pid <> pg_backend_pid()), 0),
                (select count(*) from prod.outbox_event)`);
      if (r.ok) {
        const [lag, age, total] = r.out.split("|");
        appendFileSync(`${outdir}/lag.csv`, `${now()},${lag},${age},${total}\n`);
        maxLag = Math.max(maxLag, Number(lag));
      }
    }
    if (finished) return;
    await sleep(3000);
  }
}
let finished = false;

const loops = [killLoop(), terminateLoop(), longTxLoop(), lagLoop()];

// ---- scripted schedule
const schedule: { at: number; what: () => Promise<void>; label: string }[] = [
  { at: 0.12, label: "pg graceful", what: pgGraceful },
  { at: 0.25, label: "pg crash", what: pgCrash },
  { at: 0.38, label: "replay reprocess", what: async () => {
      ev("replay (Reprocess from START) mid-run");
      for (let i = 0; i < 20; i++) {
        const r = await $`${bin} replay reprocess`.env(env).quiet().nothrow();
        if (r.exitCode === 0) { counts.replays_reprocess++; ev("replay reprocess committed"); return; }
        await sleep(1000);
      }
      ev("replay reprocess FAILED to run");
    } },
  { at: 0.5, label: "prepared tx + crash", what: preparedAcrossCrash },
  { at: 0.62, label: "pg graceful", what: pgGraceful },
  { at: 0.72, label: "replay skip", what: async () => {
      ev("replay (SkipProcessed from START) mid-run");
      for (let i = 0; i < 20; i++) {
        const r = await $`${bin} replay skip`.env(env).quiet().nothrow();
        if (r.exitCode === 0) { counts.replays_skip++; ev("replay skip committed"); return; }
        await sleep(1000);
      }
    } },
  { at: 0.8, label: "pg crash", what: pgCrash },
];
for (const s of schedule) {
  const wait = s.at * durMs - elapsed();
  if (wait > 0) await sleep(wait);
  await s.what();
}
const rest = durMs - elapsed();
if (rest > 0) await sleep(rest);

// ---- wind down
ev("chaos phase over: stopping producers gracefully, no more faults");
stopping = true;
await sleep(500);
writeFileSync(`${outdir}/STOP`, "");
// Make sure everything is running for the drain; no more kills.
await waitPg();
for (const pr of procs) if (pr.dead) { start(pr); ev(`restart ${pr.name}#${pr.n} for drain`); }
// Wait for producers to exit.
for (const pr of procs.filter((p) => p.kind === "producer")) {
  const t = setTimeout(() => pr.p?.kill("SIGKILL"), 60_000);
  await pr.p!.exited;
  clearTimeout(t);
}
// Outstanding long tx / prepared tx must be gone for the drain.
for (let i = 0; i < 60; i++) {
  const r = await psql("select count(*) from pg_stat_activity where backend_xid is not null and pid <> pg_backend_pid() and usename = 'postgres'");
  const p = await psql("select count(*) from pg_prepared_xacts");
  if (r.out === "0" && p.out === "0") break;
  await sleep(2000);
}
let drained = false;
const drainStart = Date.now();
for (let i = 0; i < 1500; i++) {
  const r = await psql(`select (select count(*) from prod.outbox_event e, cons.outbox_subscription s where (e.tx_order, e.id) > (s.position_tx, s.position_id))`);
  if (r.ok && r.out === "0") { drained = true; break; }
  await sleep(1000);
}
ev(`drain: lag reached 0 = ${drained} after ${((Date.now() - drainStart) / 1000).toFixed(1)}s`);
await sleep(4000); // one more lag sample from the library's own gauge
writeFileSync(`${outdir}/STOP_CONSUMERS`, "");
for (const pr of procs.filter((p) => p.kind === "consumer")) {
  const t = setTimeout(() => pr.p?.kill("SIGKILL"), 30_000);
  await pr.p!.exited;
  clearTimeout(t);
}
finished = true;
await Promise.allSettled(loops);

// ---- stats from poll-mode consumers (last STAT line of every incarnation: a lower bound, kills lose up to 2s)
const tot = { delivered: 0, dup: 0, dead: 0, handled: 0, retry_before: 0, retry_after: 0 };
let finalLag: string[] = [];
for (const f of readdirSync(`${outdir}/procs`)) {
  if (!f.startsWith("c") || f.endsWith(".err")) continue;
  const lines = readFileSync(`${outdir}/procs/${f}`, "utf8").split("\n").filter((l) => l.startsWith("STAT"));
  const last = lines.at(-1);
  if (!last) continue;
  for (const k of Object.keys(tot)) {
    const m = last.match(new RegExp(`${k}=(\\d+)`));
    if (m) (tot as any)[k] += Number(m[1]);
  }
  const lm = last.match(/lag=(.*)$/);
  if (f.endsWith("-" + String(procs.find((p) => f.startsWith(p.name))!.n).padStart(3, "0") + ".log")) finalLag.push(`${f}: ${lm?.[1]}`);
}
const v = await $`${bin} verify ${outdir}`.env(env).quiet().nothrow();
writeFileSync(`${outdir}/verify.json`, v.stdout.toString() + v.stderr.toString());
writeFileSync(`${outdir}/orchestrator-counts.json`, JSON.stringify({ minutes, counts, handler_side_lower_bounds: tot, max_lag_events_sampled: maxLag, final_library_lag_gauge_last_samples: finalLag, drained, verify_exit: v.exitCode }, null, 2));
console.log(v.stdout.toString());
console.log(JSON.stringify({ counts, tot, maxLag, finalLag, drained }, null, 2));
await $`podman logs ${PG}`.quiet().nothrow().then((r) => writeFileSync(`${outdir}/postgres.log`, r.stdout.toString() + r.stderr.toString()));
if (v.exitCode === 0 || process.env.KEEP_PG !== "1") await $`podman rm -f ${PG}`.quiet().nothrow();
process.exit(v.exitCode === 0 ? 0 : 1);
