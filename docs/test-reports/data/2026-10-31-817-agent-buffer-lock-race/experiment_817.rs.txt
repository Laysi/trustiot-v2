//! Laysi/trustiot-v2#817 experiments. Temporary, never committed.
#![allow(unsafe_code, reason = "experiment: pre_exec delay")]
#![allow(clippy::print_stderr, reason = "experiment output")]

use std::io::{Read, Write};
use std::os::unix::process::CommandExt;
use std::sync::Arc;
use std::sync::atomic::{AtomicBool, Ordering};
use std::time::{Duration, Instant};

use super::disk::DiskStore;

/// A child held between fork and exec keeps a copy of every descriptor, the directory's lock
/// included, after the parent dropped its store.
#[test]
fn experiment_817_a_child_between_fork_and_exec_keeps_the_lock() {
    let scratch = testkit::mqtt::ScratchDir::new("agent-exp817");
    let dir = scratch.path().join("buffer");
    let (store, _) = DiskStore::open(&dir, 1 << 20, usize::MAX).unwrap();
    let (mut forked, mut writer) = std::io::pipe().unwrap();
    let (release_reader, mut release) = std::io::pipe().unwrap();
    let spawner = std::thread::spawn(move || {
        let mut command = std::process::Command::new("true");
        // SAFETY: experiment; the closure only writes to and reads from pipes.
        unsafe {
            let mut release_reader = release_reader;
            command.pre_exec(move || {
                writer.write_all(b"f")?;
                let mut byte = [0_u8; 1];
                release_reader.read_exact(&mut byte)?;
                Ok(())
            });
        }
        command.status().unwrap()
    });
    let mut byte = [0_u8; 1];
    forked.read_exact(&mut byte).unwrap();
    drop(store);
    let early = DiskStore::open(&dir, 1 << 20, usize::MAX).map(|_| ());
    eprintln!("EXP817-A reopen while the child is between fork and exec: {early:?}");
    release.write_all(b"r").unwrap();
    let status = spawner.join().unwrap();
    let late = DiskStore::open(&dir, 1 << 20, usize::MAX).map(|_| ());
    eprintln!("EXP817-A child {status}; reopen after its exec: {late:?}");
}

/// The real path: `Command::spawn` without pre_exec (glibc posix_spawn, CLONE_VFORK) from one
/// thread while another drops a store and reopens its directory at once.
#[test]
fn experiment_817_posix_spawn_races_a_reopen() {
    let rounds: usize = std::env::var("EXP817_ROUNDS")
        .ok()
        .and_then(|value| value.parse().ok())
        .unwrap_or(20_000);
    let scratch = testkit::mqtt::ScratchDir::new("agent-exp817");
    let dir = scratch.path().join("buffer");
    let stop = Arc::new(AtomicBool::new(false));
    let spawner = {
        let stop = stop.clone();
        std::thread::spawn(move || {
            let mut spawned = 0_u64;
            while !stop.load(Ordering::Relaxed) {
                std::process::Command::new("true").status().unwrap();
                spawned += 1;
            }
            spawned
        })
    };
    let started = Instant::now();
    let mut refused = 0_u64;
    let mut longest = Duration::ZERO;
    let mut first: Option<String> = None;
    for _ in 0..rounds {
        let (store, _) = loop {
            match DiskStore::open(&dir, 1 << 20, usize::MAX) {
                Ok(opened) => break opened,
                Err(error) if error.kind() == std::io::ErrorKind::WouldBlock => {
                    std::thread::sleep(Duration::from_millis(1));
                }
                Err(error) => panic!("{error}"),
            }
        };
        drop(store);
        let reopen_started = Instant::now();
        let reopened = DiskStore::open(&dir, 1 << 20, usize::MAX);
        longest = longest.max(reopen_started.elapsed());
        match reopened {
            Ok(_) => {}
            Err(error) if error.kind() == std::io::ErrorKind::WouldBlock => {
                refused += 1;
                first.get_or_insert_with(|| error.to_string());
            }
            Err(error) => panic!("{error}"),
        }
        #[cfg(any())]
        match DiskStore::open(&dir, 1 << 20, usize::MAX) {
            Ok(_) => {}
            Err(error) if error.kind() == std::io::ErrorKind::WouldBlock => {
                refused += 1;
                first.get_or_insert_with(|| error.to_string());
            }
            Err(error) => panic!("{error}"),
        }
    }
    stop.store(true, Ordering::Relaxed);
    let spawned = spawner.join().unwrap();
    eprintln!(
        "EXP817-B {rounds} drop-then-reopen rounds in {:?} while {spawned} children were spawned: {refused} reopens refused with WouldBlock; longest reopen {longest:?}; first: {first:?}",
        started.elapsed()
    );
}
