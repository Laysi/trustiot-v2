// Builds the lab's HelmChartConfig istiod for the #255 e2e from the repo's istiod values.
// usage: bun make-hcc.ts <repo>/deploy/istio/istiod-values.yaml <live mesh yaml> <namespace> <with|without>
// The lab's existing providers (from the live mesh config) are repeated unchanged, because Helm
// replaces lists instead of merging them.
import { readFileSync } from "node:fs";

const [valuesPath, livePath, namespace, mode] = process.argv.slice(2);
if (!valuesPath || !livePath || !namespace || (mode !== "with" && mode !== "without")) {
  throw new Error("usage: make-hcc.ts <istiod-values.yaml> <live mesh yaml> <namespace> <with|without>");
}
type Provider = { name: string; envoyExtAuthzHttp: Record<string, unknown> };
const repo = Bun.YAML.parse(readFileSync(valuesPath, "utf8")) as {
  meshConfig: { pathNormalization: unknown; extensionProviders: Provider[] };
};
const live = Bun.YAML.parse(readFileSync(livePath, "utf8")) as { extensionProviders?: Provider[] };
const ours = repo.meshConfig.extensionProviders.find((p) => p.name === "trustiot-identity");
if (!ours) throw new Error("no trustiot-identity provider in the repo values");
const provider: Provider = {
  name: `trustiot-${namespace}`,
  envoyExtAuthzHttp: { ...ours.envoyExtAuthzHttp, service: `trustiot-identity.${namespace}.svc.cluster.local` },
};
const existing = (live.extensionProviders ?? []).filter((p) => p.name !== provider.name);
const meshConfig: Record<string, unknown> = { extensionProviders: [...existing, provider] };
if (mode === "with") meshConfig.pathNormalization = repo.meshConfig.pathNormalization;
const values = Bun.YAML.stringify({ meshConfig }, null, 2);
const indented = values
  .trimEnd()
  .split("\n")
  .map((line) => `    ${line.trimEnd()}`)
  .join("\n");
process.stdout.write(`# Lab only (trustiot-mac-m1), Laysi/trustiot-v2#255 e2e, generated by make-hcc.ts from
# deploy/istio/istiod-values.yaml (${mode} its meshConfig.pathNormalization): the provider
# "${provider.name}" points the repo's provider settings at the identity double in namespace
# ${namespace}; the lab's existing providers are repeated unchanged (Helm replaces lists).
#
# Apply:     sudo k3s kubectl apply -f <this file>
# Roll back: sudo k3s kubectl -n kube-system delete helmchartconfig istiod
apiVersion: helm.cattle.io/v1
kind: HelmChartConfig
metadata:
  name: istiod
  namespace: kube-system
spec:
  valuesContent: |-
${indented}
`);
