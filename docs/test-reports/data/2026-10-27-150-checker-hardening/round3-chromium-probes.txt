<script>
const props = ["opacity","margin-left","rotate","transform","padding-left","border-left-width","vertical-align","filter","text-decoration-line","float","color","font-size","visibility","display","content"];
const out = [];
for (const id of ["d","b","i","p","dc","g","t"]) {
  const el = document.getElementById(id);
  for (const pseudo of [null, "::first-letter", "::first-line", "::selection"]) {
    const s = getComputedStyle(el, pseudo);
    out.push(id + " " + (pseudo ?? "-") + " len=" + s.length + " " + props.map((p) => p + "=" + JSON.stringify(s.getPropertyValue(p))).join(" "));
  }
}
document.getElementById("out").textContent = out.join("\n");
</script>
