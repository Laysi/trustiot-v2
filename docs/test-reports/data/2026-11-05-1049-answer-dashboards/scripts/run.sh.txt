#!/usr/bin/env bash
# One-off runner for the Laysi/trustiot-v2#1049 end-to-end check, from the repository root of a
# copy that holds services/assistant/tests/spa_e2e_1049.rs and frontend/.preview-1049/.
set -uo pipefail
root=$PWD
out=$root/.e2e-1049/out
e2e=$root/.e2e-1049/run
rm -rf "$root/.e2e-1049"
mkdir -p "$out" "$e2e"

bun install --cwd frontend --frozen-lockfile > "$out/install.log" 2>&1 || { cat "$out/install.log"; exit 1; }
(cd frontend && bun --bun vite build --config .preview-1049/vite.config.ts) > "$out/preview-build.log" 2>&1 \
  || { cat "$out/preview-build.log"; exit 1; }
echo "== preview build"
grep -E "dist/" "$out/preview-build.log" | tail -30
mkdir -p "$e2e/pw"
cp -rL frontend/node_modules/playwright-core "$e2e/pw/"

echo "== harness"
E2E_DIR=$e2e cargo test --locked -p assistant --test spa_e2e_1049 -- --ignored --nocapture > "$out/harness.log" 2>&1 &
harness=$!
until [ -f "$e2e/api" ]; do
  if ! kill -0 "$harness" 2> /dev/null; then
    echo "the harness ended before it served the API"
    tail -60 "$out/harness.log"
    exit 1
  fi
  sleep 2
done
echo "API up at $(cat "$e2e/api")"

E2E_DIR=$e2e PORT=28911 bun frontend/.preview-1049/e2e-serve.ts > "$out/serve.log" 2>&1 &
serve=$!
for _ in $(seq 1 120); do curl -fsS -o /dev/null http://127.0.0.1:28911/ && break; sleep 1; done

image=$(bun frontend/.preview-1049/image.ts)
echo "== browser ($image)"
# This container engine has no host network: the container does not share the host's loopback, so it reaches the preview
# server through host.docker.internal (e2e.mjs forwards its own 127.0.0.1:28911 there).
docker run --rm --add-host host.docker.internal:host-gateway --shm-size 1g --cap-drop ALL \
  --security-opt no-new-privileges \
  -v "$root/frontend/.preview-1049:/work:ro" -v "$e2e/pw:/pw:ro" -v "$e2e:/e2e" -v "$out:/out" \
  -w /work "$image" node e2e.mjs
browser_rc=$?
touch "$e2e/done"
wait "$harness"
harness_rc=$?
kill "$serve" 2> /dev/null
echo "== proxied API requests"
cat "$out/serve.log"
echo "== harness tail"
grep -vE "^\s*(Compiling|Downloaded|Downloading)" "$out/harness.log" | tail -15
echo "== result json"
cat "$out/e2e-result.json"
echo "browser=$browser_rc harness=$harness_rc"
[ "$browser_rc" = 0 ] && [ "$harness_rc" = 0 ]
