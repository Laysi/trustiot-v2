// Scratch for Laysi/trustiot-v2#1051 (not committed): builds run.ts and ingress.ts of this
// directory from scripts/browser-e2e/, adding the assets service, so the browser test can list
// devices from the real assets API. Run: bun scripts/browser-e2e-1051/make.ts
import { readFileSync, writeFileSync } from "node:fs";
import { join } from "node:path";

const here = import.meta.dir;
const orig = join(here, "..", "browser-e2e");

function patch(text: string, pairs: [string, string][]): string {
  let out = text;
  for (const [from, to] of pairs) {
    if (!out.includes(from)) throw new Error(`missing: ${from.slice(0, 60)}`);
    out = out.replace(from, to);
  }
  return out;
}

writeFileSync(
  join(here, "ingress.ts"),
  patch(readFileSync(join(orig, "ingress.ts"), "utf8"), [
    [`  web: string;`, `  web: string;\n  /** The assets service's API origin. */\n  assets: string;`],
    [`const COOKIE_PATHS = ["/api/identity"];`, `const COOKIE_PATHS = ["/api/identity", "/api/assets"];`],
    [
      `          forwarded.set("x-internal-identity", check.headers.get("x-internal-identity") ?? "");
          return respond(await pass(options.identity + url.pathname + url.search));`,
      `          forwarded.set("x-internal-identity", check.headers.get("x-internal-identity") ?? "");
          const origin = url.pathname.startsWith("/api/assets") ? options.assets : options.identity;
          return respond(await pass(origin + url.pathname + url.search));`,
    ],
  ]),
);
