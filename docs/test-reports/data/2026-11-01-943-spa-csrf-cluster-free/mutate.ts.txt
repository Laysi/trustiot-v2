// Mutation runner: applies one textual mutation at a time to a file under `root`, runs the given
// bun test file, records whether the suite failed (the mutation was caught), then restores it.
// Usage: bun mutate.ts <root> <mutations.json>
import { readFileSync, writeFileSync } from "node:fs";
import { join } from "node:path";

type Mutation = { id: string; file: string; from: string; to: string; test: string; filter?: string; why: string };
const [root, list] = [process.argv[2]!, process.argv[3]!];
const mutations = JSON.parse(readFileSync(list, "utf8")) as Mutation[];
for (const m of mutations) {
  const path = join(root, m.file);
  const original = readFileSync(path, "utf8");
  if (original.split(m.from).length !== 2) {
    console.log(`${m.id}\tERROR\tpattern not found exactly once`);
    continue;
  }
  writeFileSync(path, original.replace(m.from, m.to));
  try {
    const args = [process.execPath, "test", m.test, ...(m.filter ? ["-t", m.filter] : [])];
    const result = Bun.spawnSync(args, { cwd: root, stdout: "pipe", stderr: "pipe", timeout: 300_000 });
    const out = `${result.stdout.toString()}${result.stderr.toString()}`;
    const fail = /^\s*(\d+) fail/m.exec(out)?.[1] ?? "?";
    const pass = /^\s*(\d+) pass/m.exec(out)?.[1] ?? "?";
    const firstFailure = out.split("\n").find((line) => line.includes("(fail)"))?.trim() ?? "";
    console.log(`${m.id}\t${result.exitCode === 0 ? "SURVIVED" : "KILLED"}\tpass=${pass} fail=${fail}\t${m.why}\t${firstFailure}`);
  } finally {
    writeFileSync(path, original);
  }
}
