// Minimal-grant checks (outbox README "資料庫權限") for the four roles of decision 29, against a
// throwaway Postgres set up exactly as the chaos run sets it up (same `chaos setup`).
// Usage: bun grants.ts <chaos-binary> <outdir>
import { $ } from "bun";
import { RUNTIME, startRun } from "./common.ts";

const [bin, outdir] = process.argv.slice(2);
const run = await startRun({ bin, outdir, name: "grants", port: Number(process.env.CHAOS_PORT ?? "55473"), seed: 1, logFile: "grants-timeline.txt" });
const { pg, password } = run;
async function sql(user: string, q: string) {
  const r = await $`${RUNTIME} exec -e PGPASSWORD=${password} ${pg} psql -h 127.0.0.1 -U ${user} -d postgres -At -v ON_ERROR_STOP=1 -v VERBOSITY=verbose -c ${q}`.quiet().nothrow();
  return { ok: r.exitCode === 0, out: r.stdout.toString().trim(), err: r.stderr.toString().trim().split("\n")[0] };
}
// A second producer the consumer was never granted anything on.
await sql("postgres", "create schema if not exists prod2; create table if not exists prod2.outbox_event (id int, envelope jsonb); insert into prod2.outbox_event values (1, '{}'); create table if not exists prod2.customers (id int, secret text); insert into prod2.customers values (1, 'x');");
const T1 = "01928f6e-7a3c-7d4e-8f00-000000000010", T2 = "01928f6e-7a3c-7d4e-8f00-000000000011";
const event = (tenant: string | null) => `insert into prod.outbox_event (tx_order, event_id, event_type, tenant_id, aggregate_type, aggregate_id, aggregate_seq, envelope) values (1, gen_random_uuid(), 't', ${tenant ? `'${tenant}'` : "null"}, 'widget', gen_random_uuid()::text, 1, '{}')`;
const inTenant = (tenant: string, stmt: string) => `begin; select set_config('trustiot.tenant_id', '${tenant}', true); ${stmt}; commit`;
type Case = { who: string; q: string; expect: "ok" | "denied"; label: string };
const CP = "chaos_cons_platform", CA = "chaos_cons_app", PA = "chaos_prod_app", PP = "chaos_prod_platform";
const cases: Case[] = [
  // The consuming service's platform role: reads the log (across tenants), nothing else of the producer.
  { who: CP, q: "select count(*) >= 0 from prod.outbox_event", expect: "ok", label: "consumer platform role reads the producer's outbox_event (consumer_read policy)" },
  { who: CP, q: "select count(*) from prod.ledger", expect: "denied", label: "consumer platform role reads producer business table" },
  { who: CP, q: "select count(*) from prod.outbox_aggregate", expect: "denied", label: "consumer platform role reads producer outbox_aggregate" },
  { who: CP, q: "select count(*) from prod2.outbox_event", expect: "denied", label: "consumer platform role reads another producer's outbox_event" },
  { who: CP, q: "select count(*) from prod2.customers", expect: "denied", label: "consumer platform role reads another producer's business table" },
  { who: CP, q: "select count(*) from secret.tenants", expect: "denied", label: "consumer platform role reads unrelated schema" },
  { who: CP, q: "select count(*) from other.junk", expect: "denied", label: "consumer platform role reads other.junk" },
  { who: CP, q: event(null), expect: "denied", label: "consumer platform role INSERTs a platform-wide event" },
  { who: CP, q: inTenant(T1, event(T1)), expect: "denied", label: "consumer platform role INSERTs a tenant event" },
  { who: CP, q: "update prod.outbox_event set event_type = 'x' where false", expect: "denied", label: "consumer platform role UPDATE producer outbox_event (zero rows)" },
  { who: CP, q: "delete from prod.outbox_event where false", expect: "denied", label: "consumer platform role DELETE producer outbox_event (zero rows)" },
  { who: CP, q: "truncate prod.outbox_event", expect: "denied", label: "consumer platform role TRUNCATE producer outbox_event" },
  { who: CP, q: "select id from prod.outbox_event limit 1 for update", expect: "denied", label: "consumer platform role SELECT FOR UPDATE producer outbox_event" },
  { who: CP, q: "insert into prod.outbox_aggregate values ('widget','zzz',1,1)", expect: "denied", label: "consumer platform role INSERT producer outbox_aggregate (cannot call publish on producer)" },
  { who: CP, q: "create table prod.x (a int)", expect: "denied", label: "consumer platform role CREATE in producer schema" },
  { who: CP, q: "select count(*) from cons.outbox_inbox", expect: "denied", label: "consumer platform role reads the consumer's own tables (they belong to the service role)" },
  { who: CP, q: "select count(*) >= 0 from pg_catalog.pg_class", expect: "ok", label: "(control) catalog readable" },
  // The consuming service's service role: its own schema only, never the producer's.
  { who: CA, q: "select count(*) from prod.outbox_event", expect: "denied", label: "consumer service role reads the producer's outbox_event" },
  { who: CA, q: "select count(*) from prod.ledger", expect: "denied", label: "consumer service role reads producer business table" },
  { who: CA, q: "select count(*) from secret.tenants", expect: "denied", label: "consumer service role reads unrelated schema" },
  { who: CA, q: "select count(*) from other.junk", expect: "denied", label: "consumer service role reads other.junk" },
  { who: CA, q: "select count(*) >= 0 from cons.outbox_inbox", expect: "ok", label: "consumer service role reads own outbox_inbox" },
  { who: CA, q: "insert into cons.outbox_dead_letter (subscription, event_id, position_tx, position_id, reason) select 's', gen_random_uuid(), 1, 1, 'r' where false", expect: "ok", label: "consumer service role writes own outbox_dead_letter" },
  { who: CA, q: "create table cons.x (a int)", expect: "denied", label: "consumer service role CREATE in own schema (not owner)" },
  { who: CA, q: "select count(*) from cons.tenant_rows", expect: "ok", label: "consumer service role reads tenant_rows (RLS: no tenant set, sees no row)" },
  { who: CA, q: `begin; select set_config('trustiot.tenant_id', '${T1}', true); insert into cons.tenant_rows values (0, gen_random_uuid(), '${T2}'); commit`, expect: "denied", label: "consumer service role writes tenant_rows of another tenant (RLS WITH CHECK)" },
  { who: CA, q: `begin; select set_config('trustiot.tenant_id', '${T1}', true); insert into cons.tenant_rows values (0, gen_random_uuid(), '${T1}'); commit`, expect: "ok", label: "consumer service role writes tenant_rows of the transaction's tenant" },
  // The producing service's roles.
  { who: PA, q: "select count(*) from prod.outbox_event", expect: "denied", label: "producer service role cannot SELECT its own outbox_event (INSERT only)" },
  { who: PA, q: inTenant(T1, event(T1)), expect: "ok", label: "producer service role INSERTs an event of the transaction's tenant" },
  { who: PA, q: inTenant(T1, event(T2)), expect: "denied", label: "producer service role INSERTs an event of another tenant (RLS WITH CHECK)" },
  { who: PA, q: event(null), expect: "denied", label: "producer service role INSERTs a platform-wide event" },
  { who: PA, q: "update prod.outbox_event set event_type = 'x' where false", expect: "denied", label: "producer service role UPDATE outbox_event" },
  { who: PA, q: "delete from prod.outbox_event where false", expect: "denied", label: "producer service role DELETE outbox_event" },
  { who: PA, q: "select count(*) from cons.outbox_inbox", expect: "denied", label: "producer service role reads consumer tables" },
  { who: PA, q: "select count(*) from secret.tenants", expect: "denied", label: "producer service role reads unrelated schema" },
  { who: PA, q: "select count(*) >= 0 from prod.outbox_aggregate", expect: "ok", label: "producer service role reads own outbox_aggregate" },
  { who: PP, q: event(null), expect: "ok", label: "producer platform role INSERTs a platform-wide event" },
  { who: PP, q: event(T1), expect: "denied", label: "producer platform role INSERTs an event of a tenant without setting the tenant" },
  // Not a defect: the template's `tenant_isolation` policy applies to every role, and the platform role
  // is cross-tenant by design (PlatformDb never sets a tenant; a role that sets one itself passes the check).
  { who: PP, q: inTenant(T1, event(T1)), expect: "ok", label: "producer platform role INSERTs an event of a tenant after setting that tenant itself (observation)" },
  { who: PP, q: "select count(*) from prod.outbox_event", expect: "denied", label: "producer platform role cannot SELECT outbox_event" },
];
let bad = 0;
const lines: string[] = [];
for (const c of cases) {
  const r = await sql(c.who, c.q);
  const got = r.ok ? "ok" : /42501|permission denied/.test(r.err) ? "denied" : "other-error";
  const pass = got === c.expect;
  if (!pass) bad++;
  lines.push(`${pass ? "PASS" : "FAIL"}  ${c.who.padEnd(20)} expect=${c.expect.padEnd(6)} got=${got.padEnd(6)} ${c.label}  [${r.ok ? r.out.split("\n")[0] : r.err}]`);
}
console.log(lines.join("\n"));
console.log(`\n${cases.length - bad}/${cases.length} as expected`);
await $`${RUNTIME} rm -f ${pg}`.quiet().nothrow();
process.exit(bad ? 1 : 0);
