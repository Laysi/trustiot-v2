//! Chaos harness for the outbox under decision 29's connection model (re-run of the harness of
//! Laysi/trustiot-v2#54's independent verification for Laysi/trustiot-v2#271). Throwaway probe,
//! not part of the product: it is copied to `crates/outbox/examples/chaos.rs` for a run and never
//! committed as Rust. Driven by `chaos.ts`, which starts, kills and restarts these processes.
//!
//! Connections: TLS with `verify-full` against a CA the orchestrator generates (`certs`), the four
//! roles of decision 29, the RLS template `crates/service-kit/sql/outbox-event.sql`, and
//! service-kit pools everywhere (`PlatformDb`, `BackgroundTenantDb`), so the producers write
//! tenant events in tenant transactions of the producer's service role and platform-wide events
//! as its platform role, and `Consumer::new(&PlatformDb, BackgroundTenantDb, ..)` reads as the
//! consuming service's platform role and handles each event as its service role in the event's
//! tenant.
//!
//! Environment: CHAOS_HOST, CHAOS_PORT, CHAOS_CA_FILE, CHAOS_PASSWORD (one password for the
//! superuser and every role of this run), CHAOS_PROC_SEED (per process, for the random choices).
//! Optional handler-panic scenario (Laysi/trustiot-v2#361; off by default, which leaves a run as
//! it was): CHAOS_PANICS=1 on the producers makes 0.3% of the events always panic in the handler
//! (dead-lettered as `handler panicked repeatedly` after three panics in a row in one replica);
//! CHAOS_PANIC_RATE=0.01 on the consumers makes the handler panic at random after its writes.
//!
//! Commands:
//!   certs <dir>                 write ca.pem, server.crt and server.key for 127.0.0.1
//!   setup                       create schemas, roles, tables, RLS and the grants
//!   producer <name> <dir>       write events to many aggregates of several tenants (and
//!                               platform-wide), some transactions rolled back
//!   consumer <name> <mode> <dir>  mode `run` (Consumer::run) or `poll` (poll_once loop with stats)
//!   replay <reprocess|skip>     move the subscription back to the start
//!   verify <dir>                check the invariants, print a JSON summary, exit 1 on failure

// The four roles of decision 29: a service role and a platform role for the producing service
// and for the consuming service.
const PROD_APP: &str = "chaos_prod_app";
const PROD_PLATFORM: &str = "chaos_prod_platform";
const CONS_APP: &str = "chaos_cons_app";
const CONS_PLATFORM: &str = "chaos_cons_platform";

const TENANTS: u64 = 5;
const AGGREGATES: u64 = 24;
const PLATFORM_AGGREGATES: u64 = 3;

/// The tenants of this run (fixed, so every process and `verify` agree).
fn tenant_text(index: u64) -> String {
    format!("01928f6e-7a3c-7d4e-8f00-0000000000{:02x}", 0x10 + index)
}

/// The tenant of tenant aggregate `a{n}`.
fn tenant_index_of(aggregate_number: u64) -> u64 {
    aggregate_number % TENANTS
}

#[derive(Serialize, Deserialize)]
struct Chaos {
    token: String,
    agg: String,
    poison: bool,
    /// Optional scenario (CHAOS_PANICS=1 on the producers): the handler panics on every attempt
    /// at this event, so the consumer must dead-letter it after `MAX_HANDLER_PANICS` in a row.
    #[serde(default)]
    panics: bool,
}

impl Event for Chaos {
    const TYPE: &'static str = "chaos.touched";
    const VERSION: u32 = 1;
    const PRODUCER: ServiceName = ServiceName::Assets;
}

fn env(name: &str) -> String {
    std::env::var(name).unwrap_or_else(|_| panic!("{name} not set"))
}

/// Connection options as `role`: TLS, `verify-full` against the run's CA.
fn options(role: &str) -> PgConnectOptions {
    PgConnectOptions::new_without_pgpass()
        .host(&env("CHAOS_HOST"))
        .port(env("CHAOS_PORT").parse().unwrap())
        .username(role)
        .password(&env("CHAOS_PASSWORD"))
        .database("postgres")
        .ssl_mode(PgSslMode::VerifyFull)
        .ssl_root_cert(env("CHAOS_CA_FILE"))
}

fn pool_options(max: u32) -> PgPoolOptions {
    PgPoolOptions::new()
        .max_connections(max)
        .acquire_timeout(Duration::from_secs(5))
}

fn platform_db(role: &str, max: u32) -> PlatformDb {
    PlatformDb::connect_lazy(options(role), pool_options(max)).expect("verify-full options")
}

fn work_db(role: &str, max: u32) -> BackgroundTenantDb {
    let (_, work) = TenantDb::connect_lazy_with_background(
        options(role),
        pool_options(max),
        ServiceIdentity::new(ServiceName::Assets),
    )
    .expect("verify-full options");
    work
}

async fn admin_pool(max: u32) -> PgPool {
    pool_options(max)
        .connect_with(options("postgres"))
        .await
        .unwrap()
}

/// xorshift, seeded per process from CHAOS_PROC_SEED (recorded by the orchestrator).
struct Rng(u64);
impl Rng {
    fn new(salt: u64) -> Self {
        let seed: u64 = std::env::var("CHAOS_PROC_SEED")
            .ok()
            .and_then(|s| s.parse().ok())
            .unwrap_or_else(|| {
                std::time::SystemTime::now()
                    .duration_since(std::time::UNIX_EPOCH)
                    .unwrap()
                    .as_nanos() as u64
            });
        Self::seeded(seed, salt)
    }
    fn seeded(seed: u64, salt: u64) -> Self {
        Self((seed ^ salt.wrapping_mul(0x9E37_79B9_7F4A_7C15)).wrapping_mul(0x2545_F491_4F6C_DD1D) | 1)
    }
    fn next(&mut self) -> u64 {
        let mut x = self.0;
        x ^= x << 13;
        x ^= x >> 7;
        x ^= x << 17;
        self.0 = x;
        x
    }
    fn f(&mut self) -> f64 {
        (self.next() >> 11) as f64 / (1u64 << 53) as f64
    }
    fn range(&mut self, lo: u64, hi: u64) -> u64 {
        lo + self.next() % (hi - lo + 1)
    }
}

#[tokio::main]
async fn main() {
    let args: Vec<String> = std::env::args().collect();
    match args.get(1).map(String::as_str) {
        Some("certs") => certs(PathBuf::from(&args[2])),
        Some("setup") => setup().await,
        Some("producer") => producer(&args[2], PathBuf::from(&args[3])).await,
        Some("consumer") => consumer(&args[2], &args[3], PathBuf::from(&args[4])).await,
        Some("replay") => replay(&args[2]).await,
        Some("verify") => verify(PathBuf::from(&args[2])).await,
        _ => panic!("usage: chaos certs|setup|producer|consumer|replay|verify"),
    }
}

/// A throwaway CA and a server certificate for 127.0.0.1 signed by it (the same construction as
/// the outbox tests' `support/postgres.rs`).
fn certs(dir: PathBuf) {
    use rcgen::{BasicConstraints, CertificateParams, DnType, IsCa, Issuer, KeyPair};
    std::fs::create_dir_all(&dir).unwrap();
    let ca_key = KeyPair::generate().unwrap();
    let mut params = CertificateParams::new(Vec::<String>::new()).unwrap();
    params.is_ca = IsCa::Ca(BasicConstraints::Unconstrained);
    params.distinguished_name.push(DnType::CommonName, "outbox chaos CA");
    let ca_pem = params.self_signed(&ca_key).unwrap().pem();
    let issuer = Issuer::new(params, ca_key);
    let key = KeyPair::generate().unwrap();
    let server = CertificateParams::new(vec!["127.0.0.1".to_owned()]).unwrap();
    let certificate = server.signed_by(&key, &issuer).unwrap();
    std::fs::write(dir.join("ca.pem"), ca_pem).unwrap();
    std::fs::write(dir.join("server.crt"), certificate.pem()).unwrap();
    std::fs::write(dir.join("server.key"), key.serialize_pem()).unwrap();
}

async fn setup() {
    let admin = admin_pool(1).await;
    let mut c = admin.acquire().await.unwrap();
    let sql = |s: String| sqlx::raw_sql(AssertSqlSafe(s));
    let pw = env("CHAOS_PASSWORD");
    // Decision 29's database template, once per database.
    sql(include_str!("../../service-kit/sql/database.sql").to_owned())
        .execute(&mut *c)
        .await
        .unwrap();
    sql(format!(
        "CREATE SCHEMA prod; CREATE SCHEMA cons; CREATE SCHEMA other; CREATE SCHEMA secret;
         CREATE TABLE other.junk (id serial PRIMARY KEY, at timestamptz DEFAULT now());
         CREATE TABLE secret.tenants (id int PRIMARY KEY, name text);
         INSERT INTO secret.tenants VALUES (1, 'x');
         CREATE ROLE {PROD_APP} LOGIN PASSWORD '{pw}';
         CREATE ROLE {PROD_PLATFORM} LOGIN PASSWORD '{pw}';
         CREATE ROLE {CONS_APP} LOGIN PASSWORD '{pw}';
         CREATE ROLE {CONS_PLATFORM} LOGIN PASSWORD '{pw}';
         ALTER ROLE {PROD_APP} SET search_path = prod;
         ALTER ROLE {PROD_PLATFORM} SET search_path = prod;
         ALTER ROLE {CONS_APP} SET search_path = cons;
         ALTER DEFAULT PRIVILEGES IN SCHEMA cons GRANT SELECT, INSERT, UPDATE, DELETE ON TABLES TO {CONS_APP};
         ALTER DEFAULT PRIVILEGES IN SCHEMA cons GRANT USAGE ON SEQUENCES TO {CONS_APP};"
    ))
    .execute(&mut *c)
    .await
    .unwrap();
    // The producer's migration: install_producer, business table, then the RLS template.
    sql("SET search_path = prod".to_owned()).execute(&mut *c).await.unwrap();
    outbox::schema::install_producer(&mut c).await.unwrap();
    sql("CREATE TABLE ledger (id bigserial PRIMARY KEY, agg text NOT NULL, token text NOT NULL UNIQUE, tenant text);"
        .to_owned())
    .execute(&mut *c)
    .await
    .unwrap();
    let template = include_str!("../../service-kit/sql/outbox-event.sql")
        .replace("example_app", PROD_APP)
        .replace("example_platform", PROD_PLATFORM)
        .replace("example_reader", CONS_PLATFORM);
    sql(template).execute(&mut *c).await.unwrap();
    // The consumer's migration.
    sql("SET search_path = cons".to_owned()).execute(&mut *c).await.unwrap();
    outbox::schema::install_consumer(&mut c).await.unwrap();
    sql(format!(
        "CREATE TABLE handled_log (n bigserial PRIMARY KEY, gen int NOT NULL, event_id uuid NOT NULL,
            agg text NOT NULL, aseq bigint NOT NULL, token text NOT NULL, replica text NOT NULL,
            setting_tenant text NOT NULL, delivery_tenant text NOT NULL);
         CREATE TABLE gen (g int NOT NULL);
         INSERT INTO gen VALUES (0);
         CREATE TABLE aggregate_state (agg text PRIMARY KEY, last_seq bigint NOT NULL, n bigint NOT NULL);
         CREATE TABLE violations (n bigserial PRIMARY KEY, kind text NOT NULL, event_id uuid, agg text, aseq bigint, last_seq bigint, detail text);
         -- Tenant data: the handler writes one row per event of a tenant; row-level security
         -- limits every query to the transaction's tenant.
         CREATE TABLE tenant_rows (gen int NOT NULL, event_id uuid NOT NULL, tenant_id uuid NOT NULL, PRIMARY KEY (gen, event_id));
         ALTER TABLE tenant_rows ENABLE ROW LEVEL SECURITY;
         CREATE POLICY tenant_isolation ON tenant_rows
             USING (tenant_id = NULLIF(current_setting('trustiot.tenant_id', true), '')::uuid)
             WITH CHECK (tenant_id = NULLIF(current_setting('trustiot.tenant_id', true), '')::uuid);
         CREATE TABLE tenant_probes (n bigserial PRIMARY KEY, gen int NOT NULL, event_id uuid NOT NULL,
            kind text NOT NULL, outcome text NOT NULL, expected text NOT NULL);"
    ))
    .execute(&mut *c)
    .await
    .unwrap();
    // The outbox README's grants (the template granted INSERT/SELECT on outbox_event itself).
    sql(format!(
        "GRANT USAGE ON SCHEMA prod TO {PROD_APP}, {PROD_PLATFORM};
         GRANT SELECT, INSERT, UPDATE ON prod.outbox_aggregate TO {PROD_APP}, {PROD_PLATFORM};
         GRANT SELECT, INSERT ON prod.ledger TO {PROD_APP}, {PROD_PLATFORM};
         GRANT USAGE ON SEQUENCE prod.ledger_id_seq TO {PROD_APP}, {PROD_PLATFORM};
         GRANT USAGE ON SCHEMA prod TO {CONS_PLATFORM};
         GRANT USAGE ON SCHEMA cons TO {CONS_APP};"
    ))
    .execute(&mut *c)
    .await
    .unwrap();
    println!("setup done");
}

fn log_line(dir: &PathBuf, name: &str, line: &str) {
    let mut f = std::fs::OpenOptions::new()
        .create(true)
        .append(true)
        .open(dir.join(name))
        .unwrap();
    f.write_all(format!("{line}\n").as_bytes()).unwrap();
}

/// An open producer transaction: a tenant transaction of the producer's service role, or a
/// transaction of its platform role for platform-wide events.
enum Open {
    Tenant(TenantTransaction),
    Platform(PlatformTransaction),
}

impl Open {
    fn tx(&mut self) -> &mut Transaction<'static, Postgres> {
        match self {
            Self::Tenant(t) => t.transaction(),
            Self::Platform(p) => p.transaction(),
        }
    }
    async fn commit(self) -> Result<(), sqlx::Error> {
        match self {
            Self::Tenant(t) => t.commit().await,
            Self::Platform(p) => p.commit().await,
        }
    }
    async fn rollback(self) -> Result<(), sqlx::Error> {
        match self {
            Self::Tenant(t) => t.rollback().await,
            Self::Platform(p) => p.rollback().await,
        }
    }
}

fn protocol(e: impl std::fmt::Display) -> sqlx::Error {
    sqlx::Error::Protocol(e.to_string())
}

async fn producer(name: &str, dir: PathBuf) {
    let work = work_db(PROD_APP, 6);
    let platform = platform_db(PROD_PLATFORM, 3);
    let incarnation = uuid::Uuid::now_v7().simple().to_string();
    let file = format!("producer-{name}-{}.log", &incarnation[..12]);
    let committed = Arc::new(AtomicU64::new(0));
    let mut tasks = Vec::new();
    for t in 0..3u64 {
        let (work, platform) = (work.clone(), platform.clone());
        let dir = dir.clone();
        let file = file.clone();
        let committed = committed.clone();
        tasks.push(tokio::spawn(async move {
            let mut rng = Rng::new(t + 1);
            let pause = std::env::var("CHAOS_PRODUCER_PAUSE_MS")
                .ok()
                .and_then(|s| s.parse::<u64>().ok())
                .unwrap_or(250);
            loop {
                if dir.join("STOP").exists() {
                    return;
                }
                tokio::time::sleep(Duration::from_millis(rng.range(pause * 2 / 5, pause * 8 / 5))).await;
                let events = match rng.range(0, 9) {
                    0..=6 => 1,
                    7..=8 => 2,
                    _ => 3,
                };
                let rollback = rng.f() < 0.12;
                let early_xid = rng.f() < 0.35;
                // 6% of the transactions write platform-wide events (kind `catalog`, tenant
                // NULL) as the producer's platform role; the others write events of one tenant
                // as its service role, in a tenant transaction.
                let platform_wide = rng.f() < 0.06;
                // The kind of each event: 0 normal, 1 poison (3%, permanent handler error), 2
                // always-panicking (0.3%, only with CHAOS_PANICS=1; drawn only then, so the
                // random sequence of a run without the scenario is unchanged).
                let panics_enabled = std::env::var("CHAOS_PANICS").is_ok_and(|v| v == "1");
                let mut kind_of = |rng: &mut Rng| -> u8 {
                    if rng.f() < 0.03 {
                        1
                    } else if panics_enabled && rng.f() < 0.003 {
                        2
                    } else {
                        0
                    }
                };
                let mut plan: Vec<(String, String, u8, Option<u64>)> = Vec::new();
                if platform_wide {
                    for _ in 0..events {
                        let a = rng.range(0, PLATFORM_AGGREGATES - 1);
                        let k = kind_of(&mut rng);
                        plan.push((format!("g{a}"), uuid::Uuid::now_v7().to_string(), k, None));
                    }
                } else {
                    // Hot aggregates: the lower of two draws; the first one fixes the tenant.
                    let first = rng.range(0, AGGREGATES - 1).min(rng.range(0, AGGREGATES - 1));
                    let tenant = tenant_index_of(first);
                    for i in 0..events {
                        let a = if i == 0 {
                            first
                        } else {
                            loop {
                                let a = rng.range(0, AGGREGATES - 1).min(rng.range(0, AGGREGATES - 1));
                                if tenant_index_of(a) == tenant {
                                    break a;
                                }
                            }
                        };
                        let k = kind_of(&mut rng);
                        plan.push((format!("a{a:02}"), uuid::Uuid::now_v7().to_string(), k, Some(tenant)));
                    }
                }
                let result: Result<bool, sqlx::Error> = async {
                    let mut open = match plan[0].3 {
                        Some(tenant) => {
                            let id: TenantId = tenant_text(tenant).parse().map_err(protocol)?;
                            Open::Tenant(work.begin(id, "chaos-publish").await.map_err(protocol)?)
                        }
                        None => Open::Platform(
                            platform
                                .begin(PlatformActor::service(ServiceIdentity::new(ServiceName::Assets)), "chaos-publish")
                                .await
                                .map_err(protocol)?,
                        ),
                    };
                    if early_xid {
                        // Take the transaction ID first, write later: IDs invert against commit order.
                        sqlx::query("SELECT pg_current_xact_id()::text")
                            .execute(&mut **open.tx())
                            .await?;
                        tokio::time::sleep(Duration::from_millis(rng.range(0, 60))).await;
                    }
                    for (agg, token, event_kind, tenant) in &plan {
                        let kind = if tenant.is_some() { "widget" } else { "catalog" };
                        let aggregate = Aggregate::new(kind, agg.clone()).unwrap();
                        let tenant_id: Option<TenantId> = match tenant {
                            Some(i) => Some(tenant_text(*i).parse().map_err(protocol)?),
                            None => None,
                        };
                        publish(
                            open.tx(),
                            &aggregate,
                            &Chaos {
                                token: token.clone(),
                                agg: agg.clone(),
                                poison: *event_kind == 1,
                                panics: *event_kind == 2,
                            },
                            tenant_id,
                        )
                        .await
                        .map_err(|e| sqlx::Error::Protocol(e.to_string()))?;
                        // The ledger row comes after the aggregate lock (taken by publish), so ledger order is commit order.
                        sqlx::query("INSERT INTO ledger (agg, token, tenant) VALUES ($1, $2, $3)")
                            .bind(agg)
                            .bind(token)
                            .bind(tenant.map(tenant_text))
                            .execute(&mut **open.tx())
                            .await?;
                        if rng.f() < 0.1 {
                            tokio::time::sleep(Duration::from_millis(rng.range(0, 30))).await;
                        }
                    }
                    if rollback {
                        for (_, token, _, _) in &plan {
                            log_line(&dir, &file, &format!("R {token}"));
                        }
                        open.rollback().await?;
                        return Ok(false);
                    }
                    for (_, token, _, _) in &plan {
                        log_line(&dir, &file, &format!("P {token}"));
                    }
                    open.commit().await?;
                    for (_, token, _, _) in &plan {
                        log_line(&dir, &file, &format!("C {token}"));
                    }
                    Ok(true)
                }
                .await;
                match result {
                    Ok(true) => {
                        committed.fetch_add(plan.len() as u64, Ordering::Relaxed);
                    }
                    Ok(false) => {}
                    Err(e) => {
                        log_line(&dir, &file, &format!("E {}", e.to_string().replace('\n', " ")));
                        tokio::time::sleep(Duration::from_millis(500)).await;
                    }
                }
            }
        }));
    }
    for t in tasks {
        let _ = t.await;
    }
    println!("producer {name} stopped cleanly, committed {}", committed.load(Ordering::Relaxed));
}

struct Counters {
    retried_before: AtomicU64,
    retried_after: AtomicU64,
    poisoned: AtomicU64,
    handled: AtomicU64,
    probes: AtomicU64,
    /// Panics raised on purpose (the optional handler-panic scenario).
    panicked: AtomicU64,
    /// Handler calls so far in this process: failure injection differs on every attempt, so a
    /// retried event is not doomed to fail again.
    attempts: AtomicU64,
}

struct Projection {
    replica: String,
    stats: Arc<Counters>,
    seed: u64,
    /// Optional scenario (CHAOS_PANIC_RATE on the consumers, e.g. 0.01): the handler panics at
    /// random, after its writes, before the commit; the consumer must roll back and retry. Zero (the
    /// default) draws nothing, so the random sequence is the same as without the scenario.
    panic_rate: f64,
}

fn sqlstate(error: &sqlx::Error) -> String {
    error
        .as_database_error()
        .and_then(|d| d.code())
        .map(|c| c.to_string())
        .unwrap_or_else(|| format!("no-sqlstate: {error}"))
}

impl Projection {
    /// Tries to leave the event's tenant, in a savepoint so the handler's own work is unaffected:
    /// a write of a row of another tenant must fail with 42501, and a read must not see any row
    /// of another tenant (for a platform-wide event: any row of any tenant).
    async fn probe(
        &self,
        tx: &mut Transaction<'static, Postgres>,
        gnum: i32,
        delivery: &Delivery,
        own: &str,
        rng: &mut Rng,
    ) -> Result<(), HandlerError> {
        let event_id = *delivery.envelope.id.as_uuid();
        // A tenant other than the event's (any tenant for a platform-wide event).
        let mut other = rng.range(0, TENANTS - 1);
        while !own.is_empty() && tenant_text(other) == own {
            other = (other + 1) % TENANTS;
        }
        sqlx::query("SAVEPOINT chaos_probe").execute(&mut **tx).await?;
        let write = sqlx::query("INSERT INTO tenant_rows (gen, event_id, tenant_id) VALUES ($1, $2, $3::text::uuid)")
            .bind(gnum)
            .bind(uuid::Uuid::now_v7())
            .bind(tenant_text(other))
            .execute(&mut **tx)
            .await;
        let outcome = match &write {
            Ok(_) => "WRITTEN".to_owned(),
            Err(e) => sqlstate(e),
        };
        // Whatever happened, undo it and go on in the same transaction.
        sqlx::query("ROLLBACK TO SAVEPOINT chaos_probe").execute(&mut **tx).await?;
        sqlx::query("RELEASE SAVEPOINT chaos_probe").execute(&mut **tx).await?;
        sqlx::query("INSERT INTO tenant_probes (gen, event_id, kind, outcome, expected) VALUES ($1, $2, 'write_other_tenant', $3, '42501')")
            .bind(gnum)
            .bind(event_id)
            .bind(&outcome)
            .execute(&mut **tx)
            .await?;
        let visible: i64 = sqlx::query_scalar(
            "SELECT count(*) FROM tenant_rows WHERE $1 = '' OR tenant_id::text <> $1",
        )
        .bind(own)
        .fetch_one(&mut **tx)
        .await?;
        sqlx::query("INSERT INTO tenant_probes (gen, event_id, kind, outcome, expected) VALUES ($1, $2, 'read_other_tenants', $3, '0')")
            .bind(gnum)
            .bind(event_id)
            .bind(visible.to_string())
            .execute(&mut **tx)
            .await?;
        self.stats.probes.fetch_add(1, Ordering::Relaxed);
        Ok(())
    }
}

impl Handler for Projection {
    async fn handle(
        &self,
        tx: &mut Transaction<'static, Postgres>,
        delivery: &Delivery,
    ) -> Result<(), HandlerError> {
        let event: Chaos = delivery.envelope.decode()?;
        let attempt = self.stats.attempts.fetch_add(1, Ordering::Relaxed);
        let mut rng = Rng::seeded(self.seed, (delivery.position.id() as u64) ^ (attempt << 40));
        if rng.f() < 0.03 {
            self.stats.retried_before.fetch_add(1, Ordering::Relaxed);
            return Err(HandlerError::retry("chaos: transient failure before write"));
        }
        let gnum: i32 = sqlx::query_scalar("SELECT g FROM gen").fetch_one(&mut **tx).await?;
        // The tenant this transaction actually runs in, read from the database session.
        let setting: Option<String> =
            sqlx::query_scalar("SELECT current_setting('trustiot.tenant_id', true)")
                .fetch_one(&mut **tx)
                .await?;
        let setting = setting.unwrap_or_default();
        let delivery_tenant = delivery.tenant().map(|t| t.to_string()).unwrap_or_default();
        if setting != delivery_tenant {
            sqlx::query("INSERT INTO violations (kind, event_id, agg, aseq, detail) VALUES ('tenant', $1, $2, $3, $4)")
                .bind(*delivery.envelope.id.as_uuid())
                .bind(&event.agg)
                .bind(delivery.aggregate_seq)
                .bind(format!("session tenant {setting:?}, delivery tenant {delivery_tenant:?}"))
                .execute(&mut **tx)
                .await?;
        }
        let last: Option<i64> = sqlx::query_scalar("SELECT last_seq FROM aggregate_state WHERE agg = $1")
            .bind(&event.agg)
            .fetch_optional(&mut **tx)
            .await?;
        if let Some(last) = last {
            if last >= delivery.aggregate_seq {
                sqlx::query("INSERT INTO violations (kind, event_id, agg, aseq, last_seq) VALUES ('order', $1,$2,$3,$4)")
                    .bind(*delivery.envelope.id.as_uuid())
                    .bind(&event.agg)
                    .bind(delivery.aggregate_seq)
                    .bind(last)
                    .execute(&mut **tx)
                    .await?;
            }
        }
        sqlx::query(
            "INSERT INTO handled_log (gen, event_id, agg, aseq, token, replica, setting_tenant, delivery_tenant) \
             VALUES ($1,$2,$3,$4,$5,$6,$7,$8)",
        )
        .bind(gnum)
        .bind(*delivery.envelope.id.as_uuid())
        .bind(&event.agg)
        .bind(delivery.aggregate_seq)
        .bind(&event.token)
        .bind(&self.replica)
        .bind(&setting)
        .bind(&delivery_tenant)
        .execute(&mut **tx)
        .await?;
        if let Some(tenant) = delivery.tenant() {
            // Tenant data, under row-level security: its WITH CHECK is the database's own proof
            // that the transaction runs in the event's tenant.
            sqlx::query("INSERT INTO tenant_rows (gen, event_id, tenant_id) VALUES ($1, $2, $3::text::uuid)")
                .bind(gnum)
                .bind(*delivery.envelope.id.as_uuid())
                .bind(tenant.to_string())
                .execute(&mut **tx)
                .await?;
        }
        if rng.f() < 0.05 {
            self.probe(tx, gnum, delivery, &delivery_tenant, &mut rng).await?;
        }
        if event.panics {
            // After writing, on every attempt: the writes roll back with each panic, and the
            // consumer dead-letters the event at its third panic in a row in this replica.
            self.stats.panicked.fetch_add(1, Ordering::Relaxed);
            panic!("chaos: always-panicking event");
        }
        if self.panic_rate > 0.0 && rng.f() < self.panic_rate {
            self.stats.panicked.fetch_add(1, Ordering::Relaxed);
            panic!("chaos: random handler panic after the writes");
        }
        if event.poison {
            self.stats.poisoned.fetch_add(1, Ordering::Relaxed);
            // After writing: the handler's writes must roll back with the dead-lettering.
            return Err(HandlerError::permanent("chaos: poison event"));
        }
        sqlx::query(
            "INSERT INTO aggregate_state (agg, last_seq, n) VALUES ($1,$2,1) \
             ON CONFLICT (agg) DO UPDATE SET last_seq = EXCLUDED.last_seq, n = aggregate_state.n + 1",
        )
        .bind(&event.agg)
        .bind(delivery.aggregate_seq)
        .execute(&mut **tx)
        .await?;
        if rng.f() < 0.03 {
            self.stats.retried_after.fetch_add(1, Ordering::Relaxed);
            return Err(HandlerError::retry("chaos: transient failure after write"));
        }
        if rng.f() < 0.2 {
            tokio::time::sleep(Duration::from_millis(rng.range(0, 15))).await;
        }
        self.stats.handled.fetch_add(1, Ordering::Relaxed);
        Ok(())
    }
}

async fn consumer(name: &str, mode: &str, dir: PathBuf) {
    let platform = platform_db(CONS_PLATFORM, 2);
    let work = work_db(CONS_APP, 4);
    let stats = Arc::new(Counters {
        retried_before: AtomicU64::new(0),
        retried_after: AtomicU64::new(0),
        poisoned: AtomicU64::new(0),
        handled: AtomicU64::new(0),
        probes: AtomicU64::new(0),
        panicked: AtomicU64::new(0),
        attempts: AtomicU64::new(0),
    });
    let config = ConsumerConfig::new(SUBSCRIPTION, "prod")
        .unwrap()
        .batch_size(50)
        .poll_interval(Duration::from_millis(100))
        .max_backoff(Duration::from_secs(2))
        .lag_interval(Duration::from_secs(1));
    let consumer = Consumer::new(
        &platform,
        work,
        config,
        Projection {
            replica: name.to_owned(),
            stats: stats.clone(),
            seed: std::env::var("CHAOS_PROC_SEED").ok().and_then(|s| s.parse().ok()).unwrap_or(7),
            panic_rate: std::env::var("CHAOS_PANIC_RATE").ok().and_then(|s| s.parse().ok()).unwrap_or(0.0),
        },
    );
    let stop = CancellationToken::new();
    {
        let stop = stop.clone();
        let dir = dir.clone();
        tokio::spawn(async move {
            while !dir.join("STOP_CONSUMERS").exists() {
                tokio::time::sleep(Duration::from_millis(500)).await;
            }
            stop.cancel();
        });
    }
    let dup = Arc::new(AtomicU64::new(0));
    let delivered = Arc::new(AtomicU64::new(0));
    let dead = Arc::new(AtomicU64::new(0));
    {
        let consumer = consumer.clone();
        let (dup, delivered, dead, stats, stop) =
            (dup.clone(), delivered.clone(), dead.clone(), stats.clone(), stop.clone());
        let name = name.to_owned();
        tokio::spawn(async move {
            while !stop.is_cancelled() {
                tokio::time::sleep(Duration::from_secs(2)).await;
                let lag = consumer.lag();
                println!(
                    "STAT {name} delivered={} dup={} dead={} handled={} retry_before={} retry_after={} probes={} panics={} lag={:?}",
                    delivered.load(Ordering::Relaxed),
                    dup.load(Ordering::Relaxed),
                    dead.load(Ordering::Relaxed),
                    stats.handled.load(Ordering::Relaxed),
                    stats.retried_before.load(Ordering::Relaxed),
                    stats.retried_after.load(Ordering::Relaxed),
                    stats.probes.load(Ordering::Relaxed),
                    stats.panicked.load(Ordering::Relaxed),
                    lag.map(|l| (l.events, l.oldest_age.map(|a| a.as_secs_f64())))
                );
            }
        });
    }
    if mode == "run" {
        consumer.run(stop).await.unwrap();
    } else {
        // Same loop as `process`, but with the Progress counters visible.
        loop {
            match consumer.ensure_subscription().await {
                Ok(()) => break,
                Err(_) => tokio::time::sleep(Duration::from_millis(500)).await,
            }
        }
        let monitor = {
            let consumer = consumer.clone();
            let stop = stop.clone();
            tokio::spawn(async move { consumer.monitor_lag(stop).await })
        };
        let mut failures = 0u32;
        while !stop.is_cancelled() {
            let wait = match consumer.poll_once().await {
                Ok(p) => {
                    delivered.fetch_add(p.delivered as u64, Ordering::Relaxed);
                    dup.fetch_add(p.duplicates as u64, Ordering::Relaxed);
                    dead.fetch_add(p.dead_lettered as u64, Ordering::Relaxed);
                    match p.outcome {
                        Outcome::MoreAvailable => {
                            failures = 0;
                            Duration::ZERO
                        }
                        Outcome::CaughtUp | Outcome::Busy => {
                            failures = 0;
                            Duration::from_millis(100)
                        }
                        Outcome::Retry { .. } => {
                            failures += 1;
                            Duration::from_millis(100 * 2u64.pow(failures.min(4)))
                        }
                    }
                }
                Err(_) => {
                    failures += 1;
                    Duration::from_millis(100 * 2u64.pow(failures.min(4)))
                }
            };
            tokio::time::sleep(wait).await;
        }
        let _ = monitor.await;
    }
    println!("consumer {name} stopped cleanly");
}

async fn replay(mode: &str) {
    let work = work_db(CONS_APP, 1);
    let mode = if mode == "reprocess" { Replay::Reprocess } else { Replay::SkipProcessed };
    // The consuming service's role, in a transaction without a tenant: the subscription and
    // `aggregate_state` are not tenant data.
    let mut tx = work.begin_without_tenant("chaos-replay").await.unwrap();
    outbox::replay_from(&mut tx, SUBSCRIPTION, Position::START, mode)
        .await
        .unwrap();
    if mode == Replay::Reprocess {
        sqlx::query("UPDATE gen SET g = g + 1").execute(&mut *tx).await.unwrap();
        sqlx::query("DELETE FROM aggregate_state").execute(&mut *tx).await.unwrap();
    }
    tx.commit().await.unwrap();
    println!("replay {mode:?} committed");
}

type EventRow = (i64, i64, uuid::Uuid, String, i64, String, bool, Option<String>);

async fn verify(dir: PathBuf) {
    let pool = admin_pool(2).await;
    let mut failures: Vec<String> = Vec::new();
    let mut fail = |m: String| failures.push(m);

    // Producer-side truth.
    let events: Vec<EventRow> = sqlx::query_as(
        "SELECT id, tx_order, event_id, aggregate_id, aggregate_seq, envelope->'payload'->>'token', \
         ((envelope->'payload'->>'poison')::bool OR COALESCE((envelope->'payload'->>'panics')::bool, false)), \
         tenant_id::text FROM prod.outbox_event ORDER BY id",
    )
    .fetch_all(&pool)
    .await
    .unwrap();
    // `e.6` above is "ends in the dead letters": poison events (permanent error) and, with the
    // optional handler-panic scenario, always-panicking events (dead-lettered after 3 panics).
    let panickers: HashSet<uuid::Uuid> = sqlx::query_scalar::<_, uuid::Uuid>(
        "SELECT event_id FROM prod.outbox_event WHERE COALESCE((envelope->'payload'->>'panics')::bool, false)",
    )
    .fetch_all(&pool)
    .await
    .unwrap()
    .into_iter()
    .collect();
    let ledger: Vec<(i64, String, String, Option<String>)> =
        sqlx::query_as("SELECT id, agg, token, tenant FROM prod.ledger ORDER BY id")
            .fetch_all(&pool)
            .await
            .unwrap();
    let ev_tokens: HashSet<&str> = events.iter().map(|e| e.5.as_str()).collect();
    let led_tokens: HashSet<&str> = ledger.iter().map(|e| e.2.as_str()).collect();
    if ev_tokens != led_tokens || events.len() != ledger.len() {
        fail(format!(
            "events and ledger differ: {} events, {} ledger rows, {} tokens only in events, {} only in ledger",
            events.len(),
            ledger.len(),
            ev_tokens.difference(&led_tokens).count(),
            led_tokens.difference(&ev_tokens).count()
        ));
    }
    // The producer's tenant (ledger, written by the producer in the same transaction) must be the
    // log row's tenant_id, which the producer's row-level security checked.
    let ledger_tenant: HashMap<&str, &Option<String>> =
        ledger.iter().map(|l| (l.2.as_str(), &l.3)).collect();
    let producer_tenant_mismatch = events
        .iter()
        .filter(|e| ledger_tenant.get(e.5.as_str()).is_some_and(|t| **t != e.7))
        .count();
    if producer_tenant_mismatch > 0 {
        fail(format!("{producer_tenant_mismatch} events whose log tenant differs from the producer's ledger tenant"));
    }
    // Producer logs.
    let (mut rolled, mut pending, mut acked, mut errors) = (HashSet::new(), HashSet::new(), HashSet::new(), 0usize);
    for entry in std::fs::read_dir(&dir).unwrap() {
        let path = entry.unwrap().path();
        if path.extension().and_then(|e| e.to_str()) != Some("log")
            || !path.file_name().unwrap().to_string_lossy().starts_with("producer-")
        {
            continue;
        }
        // Records are `<R|P|C> <36-char token>`; older logs may have records glued together because
        // a record was written with two write calls from concurrent tasks.
        let text = std::fs::read_to_string(&path).unwrap();
        errors += text.lines().filter(|l| l.starts_with("E ")).count();
        let bytes: Vec<char> = text.chars().collect();
        let mut i = 0;
        while i + 38 <= bytes.len() {
            if matches!(bytes[i], 'R' | 'P' | 'C') && bytes[i + 1] == ' ' && bytes[i + 2 + 8] == '-' && bytes[i + 2 + 13] == '-' {
                let token: String = bytes[i + 2..i + 38].iter().collect();
                match bytes[i] {
                    'R' => rolled.insert(token),
                    'P' => pending.insert(token),
                    _ => acked.insert(token),
                };
                i += 38;
            } else {
                i += 1;
            }
        }
    }
    let rolled_in_log = rolled.iter().filter(|t| ev_tokens.contains(t.as_str())).count();
    if rolled_in_log > 0 {
        fail(format!("{rolled_in_log} events from rolled-back transactions exist"));
    }
    let rolled_in_ledger = rolled.iter().filter(|t| led_tokens.contains(t.as_str())).count();
    if rolled_in_ledger > 0 {
        fail(format!("{rolled_in_ledger} ledger rows from rolled-back transactions exist"));
    }
    let lost_acked: Vec<&String> = acked.iter().filter(|t| !ev_tokens.contains(t.as_str())).collect();
    if !lost_acked.is_empty() {
        fail(format!("{} acknowledged commits have no event", lost_acked.len()));
    }
    let pending_missing = pending.iter().filter(|t| !ev_tokens.contains(t.as_str())).count();
    let pending_present = pending.len() - pending_missing;

    // Producer order: ledger id order per aggregate (commit order, serialized by the row lock)
    // against aggregate_seq.
    let mut seq_by_token: HashMap<&str, i64> = HashMap::new();
    for e in &events {
        seq_by_token.insert(e.5.as_str(), e.4);
    }
    let mut ledger_order: BTreeMap<&str, Vec<i64>> = BTreeMap::new();
    for (_, agg, token, _) in &ledger {
        if let Some(s) = seq_by_token.get(token.as_str()) {
            ledger_order.entry(agg.as_str()).or_default().push(*s);
        }
    }
    for (agg, seqs) in &ledger_order {
        if !seqs.windows(2).all(|w| w[0] < w[1]) {
            fail(format!("aggregate {agg}: aggregate_seq does not follow ledger commit order"));
        }
        let expected: Vec<i64> = (1..=seqs.len() as i64).collect();
        if *seqs != expected {
            fail(format!("aggregate {agg}: sequence has gaps"));
        }
    }
    // tx_order never goes down within an aggregate.
    let mut last_tx: HashMap<&str, (i64, i64)> = HashMap::new();
    let mut by_seq = events.iter().collect::<Vec<_>>();
    by_seq.sort_by_key(|e| (e.3.clone(), e.4));
    let mut tx_order_inversions = 0;
    for e in &by_seq {
        if let Some((tx, id)) = last_tx.get(e.3.as_str()) {
            if e.1 < *tx || e.0 < *id {
                tx_order_inversions += 1;
            }
        }
        last_tx.insert(e.3.as_str(), (e.1, e.0));
    }
    if tx_order_inversions > 0 {
        fail(format!("{tx_order_inversions} aggregate events sort before their predecessor"));
    }

    // Consumer side.
    let log: Vec<(i64, i32, uuid::Uuid, String, i64, String, String)> = sqlx::query_as(
        "SELECT n, gen, event_id, agg, aseq, setting_tenant, delivery_tenant FROM cons.handled_log ORDER BY n",
    )
    .fetch_all(&pool)
    .await
    .unwrap();
    let final_gen: i32 = sqlx::query_scalar("SELECT g FROM cons.gen").fetch_one(&pool).await.unwrap();
    let poison: HashSet<uuid::Uuid> = events.iter().filter(|e| e.6).map(|e| e.2).collect();
    let good: HashSet<uuid::Uuid> = events.iter().filter(|e| !e.6).map(|e| e.2).collect();
    // The tenant of each event, from the producer's log row ('' for a platform-wide event).
    let tenant_of: HashMap<uuid::Uuid, String> =
        events.iter().map(|e| (e.2, e.7.clone().unwrap_or_default())).collect();
    let mut per_gen: BTreeMap<i32, Vec<&(i64, i32, uuid::Uuid, String, i64, String, String)>> = BTreeMap::new();
    for l in &log {
        per_gen.entry(l.1).or_default().push(l);
    }
    let tenant_rows: Vec<(i32, uuid::Uuid, String)> =
        sqlx::query_as("SELECT gen, event_id, tenant_id::text FROM cons.tenant_rows")
            .fetch_all(&pool)
            .await
            .unwrap();
    let mut tenant_rows_by_gen: BTreeMap<i32, HashMap<uuid::Uuid, String>> = BTreeMap::new();
    for r in &tenant_rows {
        tenant_rows_by_gen.entry(r.0).or_default().insert(r.1, r.2.clone());
    }
    let mut gen_summary = Vec::new();
    let (mut tenant_checked, mut tenant_session_mismatch, mut tenant_delivery_mismatch) = (0usize, 0usize, 0usize);
    let (mut tenant_row_missing, mut tenant_row_wrong, mut tenant_row_platform) = (0usize, 0usize, 0usize);
    for (gnum, rows) in &per_gen {
        let mut seen = HashSet::new();
        let mut dups = 0;
        let mut last: HashMap<&str, i64> = HashMap::new();
        let mut order_violations = 0;
        for r in rows {
            if !seen.insert(r.2) {
                dups += 1;
            }
            if let Some(prev) = last.get(r.3.as_str()) {
                if *prev >= r.4 {
                    order_violations += 1;
                }
            }
            last.insert(r.3.as_str(), r.4);
            // Each event was handled in its own tenant: the session's tenant setting while the
            // handler ran, and the delivery's tenant, both equal the producer's log row.
            if let Some(expected) = tenant_of.get(&r.2) {
                tenant_checked += 1;
                if &r.5 != expected {
                    tenant_session_mismatch += 1;
                }
                if &r.6 != expected {
                    tenant_delivery_mismatch += 1;
                }
            }
        }
        if dups > 0 {
            fail(format!("generation {gnum}: {dups} events handled more than once"));
        }
        if order_violations > 0 {
            fail(format!("generation {gnum}: {order_violations} per-aggregate order violations"));
        }
        // Tenant rows (written under row-level security) of this generation: exactly one per
        // handled event of a tenant, in that tenant, none for platform-wide events.
        let rows_of_gen = tenant_rows_by_gen.get(gnum);
        for r in rows {
            let expected = &tenant_of[&r.2];
            let got = rows_of_gen.and_then(|m| m.get(&r.2));
            match (expected.is_empty(), got) {
                (true, Some(_)) => tenant_row_platform += 1,
                (true, None) => {}
                (false, None) => tenant_row_missing += 1,
                (false, Some(t)) => {
                    if t != expected {
                        tenant_row_wrong += 1;
                    }
                }
            }
        }
        let handled_ids: HashSet<uuid::Uuid> = rows.iter().map(|r| r.2).collect();
        let stray = rows_of_gen.map_or(0, |m| m.keys().filter(|k| !handled_ids.contains(k)).count());
        if stray > 0 {
            fail(format!("generation {gnum}: {stray} tenant rows of events that were not handled (a rolled-back write leaked)"));
        }
        // Poison events are written then rolled back, so they must never be in the log.
        let in_log_poison = rows.iter().filter(|r| poison.contains(&r.2)).count();
        if in_log_poison > 0 {
            fail(format!("generation {gnum}: {in_log_poison} poison events left handler writes behind"));
        }
        let unknown = rows.iter().filter(|r| !good.contains(&r.2)).count();
        if unknown > 0 {
            fail(format!("generation {gnum}: {unknown} handled events are not committed non-poison events"));
        }
        let complete = seen == good;
        if *gnum == final_gen && !complete {
            fail(format!(
                "final generation {gnum}: handled {} of {} events; {} missing",
                seen.len(),
                good.len(),
                good.difference(&seen).count()
            ));
        }
        // Per aggregate, a generation is a prefix of the aggregate's events (no skipped event).
        let mut by_agg: BTreeMap<&str, Vec<i64>> = BTreeMap::new();
        for r in rows {
            by_agg.entry(r.3.as_str()).or_default().push(r.4);
        }
        let mut non_prefix = 0;
        let mut good_by_agg: BTreeMap<&str, Vec<i64>> = BTreeMap::new();
        for e in events.iter().filter(|e| !e.6) {
            good_by_agg.entry(e.3.as_str()).or_default().push(e.4);
        }
        for v in good_by_agg.values_mut() {
            v.sort();
        }
        for (agg, got) in &by_agg {
            let want = &good_by_agg[agg];
            if got.as_slice() != &want[..got.len().min(want.len())] || got.len() > want.len() {
                non_prefix += 1;
            }
        }
        if non_prefix > 0 {
            fail(format!("generation {gnum}: {non_prefix} aggregates skipped an event or reordered"));
        }
        gen_summary.push(serde_json::json!({"gen": gnum, "handled": rows.len(), "distinct": seen.len(), "complete": complete}));
    }
    if tenant_session_mismatch > 0 {
        fail(format!("{tenant_session_mismatch} events handled in a session whose tenant differs from the event's tenant"));
    }
    if tenant_delivery_mismatch > 0 {
        fail(format!("{tenant_delivery_mismatch} deliveries whose tenant differs from the event's tenant"));
    }
    if tenant_row_missing + tenant_row_wrong + tenant_row_platform > 0 {
        fail(format!(
            "tenant rows: {tenant_row_missing} missing, {tenant_row_wrong} in the wrong tenant, {tenant_row_platform} written for platform-wide events"
        ));
    }
    // Tenant probes: a write to another tenant is refused with 42501, no other tenant's row is visible.
    let probes: Vec<(String, String, String)> =
        sqlx::query_as("SELECT kind, outcome, expected FROM cons.tenant_probes")
            .fetch_all(&pool)
            .await
            .unwrap();
    let mut probe_counts: BTreeMap<&str, (usize, usize)> = BTreeMap::new();
    for (kind, outcome, expected) in &probes {
        let entry = probe_counts.entry(kind.as_str()).or_default();
        entry.0 += 1;
        if outcome != expected {
            entry.1 += 1;
        }
    }
    for (kind, (total, wrong)) in &probe_counts {
        if *wrong > 0 {
            fail(format!("tenant probe {kind}: {wrong} of {total} did not behave as expected"));
        }
    }
    if events.len() > 2000 && probes.is_empty() {
        fail("no tenant probe ran".to_owned());
    }
    // State and bookkeeping.
    let state: Vec<(String, i64, i64)> =
        sqlx::query_as("SELECT agg, last_seq, n FROM cons.aggregate_state ORDER BY agg")
            .fetch_all(&pool)
            .await
            .unwrap();
    let mut want_state: BTreeMap<&str, (i64, i64)> = BTreeMap::new();
    for e in events.iter().filter(|e| !e.6) {
        let s = want_state.entry(e.3.as_str()).or_insert((0, 0));
        s.0 = s.0.max(e.4);
        s.1 += 1;
    }
    let got_state: BTreeMap<&str, (i64, i64)> = state.iter().map(|s| (s.0.as_str(), (s.1, s.2))).collect();
    if want_state != got_state {
        fail("consumer aggregate_state differs from the producer's events".to_owned());
    }
    let violations: Vec<(String, i64)> =
        sqlx::query_as("SELECT kind, count(*) FROM cons.violations GROUP BY kind")
            .fetch_all(&pool)
            .await
            .unwrap();
    for (kind, n) in &violations {
        fail(format!("{n} in-handler {kind} violations recorded"));
    }
    let violation_total: i64 = violations.iter().map(|v| v.1).sum();
    let dead: Vec<(uuid::Uuid, String)> = sqlx::query_as("SELECT event_id, reason FROM cons.outbox_dead_letter")
        .fetch_all(&pool)
        .await
        .unwrap();
    let dead_set: HashSet<uuid::Uuid> = dead.iter().map(|d| d.0).collect();
    // Each dead letter has the reason of its kind: a panic-looping event "handler panicked
    // repeatedly", a poison event the handler's own error.
    let mut dead_reasons: BTreeMap<String, usize> = BTreeMap::new();
    for (id, reason) in &dead {
        *dead_reasons.entry(reason.clone()).or_default() += 1;
        let expected = if panickers.contains(id) { "handler panicked repeatedly" } else { "chaos: poison event" };
        if reason != expected {
            fail(format!("dead letter {id} has reason {reason:?}, expected {expected:?}"));
        }
    }
    if dead_set != poison {
        fail(format!(
            "dead letters ({}) differ from the poison events ({}): {} missing, {} extra",
            dead_set.len(),
            poison.len(),
            poison.difference(&dead_set).count(),
            dead_set.difference(&poison).count()
        ));
    }
    let inbox: i64 = sqlx::query_scalar("SELECT count(*) FROM cons.outbox_inbox").fetch_one(&pool).await.unwrap();
    if inbox as usize != good.len() {
        fail(format!("inbox has {inbox} rows, expected {}", good.len()));
    }
    let inbox_ids: Vec<(uuid::Uuid,)> = sqlx::query_as("SELECT event_id FROM cons.outbox_inbox").fetch_all(&pool).await.unwrap();
    let inbox_set: HashSet<uuid::Uuid> = inbox_ids.iter().map(|d| d.0).collect();
    if inbox_set != good {
        fail("inbox event IDs differ from the committed non-poison events".to_owned());
    }
    let pos: (i64, i64) = sqlx::query_as("SELECT position_tx, position_id FROM cons.outbox_subscription")
        .fetch_one(&pool)
        .await
        .unwrap();
    let maxpos = events.iter().map(|e| (e.1, e.0)).max().unwrap();
    if pos != maxpos {
        fail(format!("subscription at {pos:?}, last event at {maxpos:?}"));
    }
    let lag: i64 = sqlx::query_scalar(
        "SELECT count(*) FROM prod.outbox_event WHERE (tx_order, id) > ($1, $2)",
    )
    .bind(pos.0)
    .bind(pos.1)
    .fetch_one(&pool)
    .await
    .unwrap();
    if lag != 0 {
        fail(format!("lag is {lag}, expected 0"));
    }
    let distinct_aggs: BTreeSet<&str> = events.iter().map(|e| e.3.as_str()).collect();
    // Per tenant: events written and handled in the final generation.
    let mut per_tenant: BTreeMap<String, (usize, usize)> = BTreeMap::new();
    for e in &events {
        let key = e.7.clone().unwrap_or_else(|| "platform-wide".to_owned());
        per_tenant.entry(key).or_default().0 += 1;
    }
    if let Some(rows) = per_gen.get(&final_gen) {
        for r in rows {
            let t = &tenant_of[&r.2];
            let key = if t.is_empty() { "platform-wide".to_owned() } else { t.clone() };
            per_tenant.entry(key).or_default().1 += 1;
        }
    }
    let summary = serde_json::json!({
        "events_in_outbox": events.len(),
        "ledger_rows": ledger.len(),
        "poison_events": poison.len() - panickers.len(),
        "always_panicking_events": panickers.len(),
        "dead_letter_reasons": dead_reasons,
        "rolled_back_tokens_logged": rolled.len(),
        "rolled_back_found_in_outbox": rolled_in_log,
        "committed_acks_logged": acked.len(),
        "pending_commit_tokens": pending.len(),
        "pending_commit_present": pending_present,
        "pending_commit_absent": pending_missing,
        "producer_error_lines": errors,
        "aggregates": distinct_aggs.len(),
        "tx_order_inversions": tx_order_inversions,
        "final_generation": final_gen,
        "generations": gen_summary,
        "handled_log_rows": log.len(),
        "dead_letters": dead_set.len(),
        "inbox_rows": inbox,
        "in_handler_violations": violation_total,
        "lag_events": lag,
        "tenants": {
            "events_per_tenant_and_handled_in_final_generation": per_tenant,
            "deliveries_checked_against_the_log_rows_tenant": tenant_checked,
            "session_tenant_mismatches": tenant_session_mismatch,
            "delivery_tenant_mismatches": tenant_delivery_mismatch,
            "producer_ledger_tenant_mismatches": producer_tenant_mismatch,
            "tenant_rows_missing": tenant_row_missing,
            "tenant_rows_wrong_tenant": tenant_row_wrong,
            "tenant_rows_for_platform_wide_events": tenant_row_platform,
            "tenant_rows_total": tenant_rows.len(),
            "probes": probe_counts.iter().map(|(k, v)| (k.to_string(), serde_json::json!({"total": v.0, "unexpected": v.1}))).collect::<BTreeMap<_, _>>(),
        },
        "failures": failures,
    });
    println!("{}", serde_json::to_string_pretty(&summary).unwrap());
    if !summary["failures"].as_array().unwrap().is_empty() {
        std::process::exit(1);
    }
}
