//! Throwaway probes for the independent verification of Laysi/trustiot-v2#366 (not committed).
//!
//! Each case builds a fresh database with one table `items` (row-level security, the template's
//! `tenant_isolation`, two rows of tenant A and one of tenant B), applies a setup as the
//! superuser, and then reports (1) what `check_session(.., SessionRole::Platform)` returns for a
//! session of the case's platform role and (2) what a plain session of that role (no check)
//! actually reaches with each probe statement, without a tenant and after it set tenant A itself.
//! "unsafe" means a probe reaches more rows with tenant A set than without a tenant.
#![cfg(feature = "contract")]
#![allow(clippy::print_stdout, reason = "probe output is the evidence")]

fn fill(sql: &str, n: usize) -> String {
    sql.replace("{T}", T)
        .replace("{A}", A)
        .replace("{B}", B)
        .replace("{p}", &format!("v{n}_p"))
        .replace("{g}", &format!("v{n}_g"))
        .replace("{h}", &format!("v{n}_h"))
}

async fn run(connection: &mut PgConnection, sql: String) {
    sqlx::raw_sql(AssertSqlSafe(sql.clone()))
        .execute(connection)
        .await
        .unwrap_or_else(|error| panic!("{sql}: {error}"));
}

fn short(error: &sqlx::Error) -> String {
    match error {
        sqlx::Error::Database(db) => {
            format!("ERR {} {}", db.code().unwrap_or_default(), db.message())
        }
        other => format!("ERR {other}"),
    }
}

/// Rows `sql` reaches in a plain session (no check) after an optional SET ROLE and tenant.
async fn reach(
    connection: &mut PgConnection,
    role: Option<&str>,
    tenant: Option<&str>,
    sql: &'static str,
) -> Result<u64, String> {
    let mut tx = connection.begin().await.expect("begin");
    let result: Result<u64, sqlx::Error> = async {
        if let Some(role) = role {
            sqlx::query("SELECT set_config('role', $1, true)")
                .bind(role)
                .execute(&mut *tx)
                .await?;
        }
        if let Some(tenant) = tenant {
            sqlx::query("SELECT set_config('trustiot.tenant_id', $1, true)")
                .bind(tenant)
                .execute(&mut *tx)
                .await?;
        }
        sqlx::query(sql)
            .execute(&mut *tx)
            .await
            .map(|done| done.rows_affected())
    }
    .await;
    tx.rollback().await.expect("rollback");
    result.map_err(|error| short(&error))
}

fn rows(result: &Result<u64, String>) -> u64 {
    *result.as_ref().unwrap_or(&0)
}

#[tokio::test]
async fn probes() {
    let Some(pg) = tls_postgres(INIT).await else {
        return;
    };
    let mut admin = PgConnection::connect_with(&pg.superuser("postgres"))
        .await
        .expect("superuser");
    let version: String = sqlx::query_scalar("SELECT version()")
        .fetch_one(&mut admin)
        .await
        .expect("version");
    println!("server: {version}");
    let mut findings = Vec::new();
    for (index, case) in cases().into_iter().enumerate() {
        let n = index + 1;
        run(
            &mut admin,
            format!(
                "CREATE ROLE v{n}_p LOGIN PASSWORD 'service-kit-test';
                 CREATE ROLE v{n}_g NOLOGIN; CREATE ROLE v{n}_h NOLOGIN;"
            ),
        )
        .await;
        run(&mut admin, format!("CREATE DATABASE v{n} OWNER v_owner")).await;
        let mut owner = PgConnection::connect_with(&pg.options("v_owner", &format!("v{n}")))
            .await
            .expect("owner");
        run(&mut owner, include_str!("../sql/database.sql").to_owned()).await;
        run(&mut owner, fill(BASE, n)).await;
        owner.close().await.expect("close");
        let mut superuser = PgConnection::connect_with(&pg.superuser(&format!("v{n}")))
            .await
            .expect("superuser on the case database");
        for step in case.setup {
            run(&mut superuser, fill(step, n)).await;
        }
        superuser.close().await.expect("close");

        let options = pg.options(&format!("v{n}_p"), &format!("v{n}"));
        let mut session = PgConnection::connect_with(&options)
            .await
            .expect("platform");
        let checked = postgres::check_session(&mut session, SessionRole::Platform)
            .await
            .expect("check runs");
        let mut unsafe_probe = false;
        let mut lines = Vec::new();
        for (role, sql) in case.probes {
            let role = role.map(|role| fill(role, n));
            let without = reach(&mut session, role.as_deref(), None, sql).await;
            let with = reach(&mut session, role.as_deref(), Some(A), sql).await;
            if rows(&with) > rows(&without) {
                unsafe_probe = true;
            }
            lines.push(format!(
                "    [{}] {sql}\n      no tenant: {without:?}\n      tenant A:  {with:?}",
                role.as_deref().unwrap_or("as p")
            ));
        }
        let tenant_only = matches!(checked, Err(UnsafeSession::PlatformTenantOnly { .. }));
        let verdict = match (&checked, unsafe_probe) {
            (Ok(()), true) => "FALSE NEGATIVE (unsafe, accepted)",
            (Err(_), false) if tenant_only => "FALSE POSITIVE (safe, refused PlatformTenantOnly)",
            (Err(_), false) => "refused for another reason; safe w.r.t. tenant",
            (Err(_), true) if !tenant_only => "unsafe, refused for another reason",
            (Err(_), true) => "agrees (unsafe, refused)",
            (Ok(()), false) => "agrees (safe, accepted)",
        };
        println!("\n=== case {} — {}", case.id, case.name);
        println!("  check (Platform): {checked:?}");
        for line in &lines {
            println!("{line}");
        }
        println!("  verdict: {verdict}");
        if verdict.starts_with("FALSE") {
            findings.push(format!("{} {}: {verdict}", case.id, case.name));
        }
        session.close().await.expect("close");
    }
    println!("\n=== mismatches ({}):", findings.len());
    for finding in findings {
        println!("  {finding}");
    }
}

async fn measure(pg: &TlsPostgres, role: &str, kind: SessionRole, label: &str) {
    let mut session = PgConnection::connect_with(&pg.options(role, "vcost"))
        .await
        .expect("connect");
    let first = postgres::check_session(&mut session, kind)
        .await
        .expect("check");
    let mut times: Vec<Duration> = Vec::new();
    for _ in 0..15 {
        let started = Instant::now();
        let again = postgres::check_session(&mut session, kind)
            .await
            .expect("check");
        times.push(started.elapsed());
        assert_eq!(again, first);
    }
    times.sort();
    println!(
        "cost {label} [{kind:?} {role}]: result {first:?}; median {:.1} ms, min {:.1} ms, max {:.1} ms (15 runs)",
        times[7].as_secs_f64() * 1e3,
        times[0].as_secs_f64() * 1e3,
        times[14].as_secs_f64() * 1e3
    );
    session.close().await.expect("close");
}

fn groups_sql(from: usize, to: usize) -> String {
    format!(
        "DO $$
         DECLARE i int; roles text;
         BEGIN
           SELECT string_agg(format('vc_g%s', g), ', ') INTO roles FROM generate_series({from}, {to}) g;
           EXECUTE format('GRANT %s TO vc_p', roles);
         END $$;"
    )
}

fn group_rights_sql(to: usize) -> String {
    format!(
        "DO $$
         DECLARE i int; roles text;
         BEGIN
           SELECT string_agg(format('vc_g%s', g), ', ') INTO roles FROM generate_series(1, {to}) g;
           FOR i IN 1..300 LOOP
             EXECUTE format('GRANT SELECT, INSERT, UPDATE, DELETE ON t%s TO %s', i, roles);
             EXECUTE format('ALTER POLICY platform_access ON t%s TO vc_p, %s', i, roles);
           END LOOP;
         END $$;"
    )
}

             EXECUTE format('CREATE TABLE t%s (tenant_id uuid NOT NULL, id uuid NOT NULL, name text NOT NULL, PRIMARY KEY (tenant_id, id))', i);
             EXECUTE format('ALTER TABLE t%s ENABLE ROW LEVEL SECURITY', i);
             EXECUTE format($f$CREATE POLICY tenant_isolation ON t%s USING (tenant_id = NULLIF(current_setting('trustiot.tenant_id', true), '')::uuid) WITH CHECK (tenant_id = NULLIF(current_setting('trustiot.tenant_id', true), '')::uuid)$f$, i);
             EXECUTE format('CREATE POLICY platform_access ON t%s TO vc_p USING (true) WITH CHECK (true)', i);
             EXECUTE format('GRANT SELECT, INSERT, UPDATE, DELETE ON t%s TO vc_app', i);
             EXECUTE format('GRANT SELECT, UPDATE ON t%s TO vc_p', i);
           END LOOP;
         END $$;"
            .to_owned(),
    )
    .await;
    owner.close().await.expect("close");
    let mut db_admin = PgConnection::connect_with(&pg.superuser("vcost"))
        .await
        .expect("superuser on vcost");

    measure(&pg, "vc_app", SessionRole::Service, "S0 300 tables").await;
    measure(
        &pg,
        "vc_p",
        SessionRole::Platform,
        "S0 300 tables, no groups",
    )
    .await;
    run(&mut db_admin, groups_sql(1, 20)).await;
    measure(
        &pg,
        "vc_p",
        SessionRole::Platform,
        "S1 + 20 groups without rights",
    )
    .await;
    run(&mut db_admin, group_rights_sql(20)).await;
    measure(
        &pg,
        "vc_p",
        SessionRole::Platform,
        "S2 20 groups with all rights on 300 tables, named in platform_access",
    )
    .await;
    run(&mut db_admin, groups_sql(21, 60)).await;
    run(&mut db_admin, group_rights_sql(60)).await;
    measure(
        &pg,
        "vc_p",
        SessionRole::Platform,
        "S3 60 groups with all rights on 300 tables, named in platform_access",
    )
    .await;
    measure(&pg, "vc_app", SessionRole::Service, "S3 service role").await;
