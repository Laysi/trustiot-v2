// Renders both nightly shapes with the smoke's values, the API values and both certs values files
// (#517), and validates the result with kubeconform at the chart's two Kubernetes versions.
// Usage: bun render.ts <worktree> <outdir>
import { mkdirSync, mkdtempSync, readFileSync, rmSync, writeFileSync } from "node:fs";
import { join } from "node:path";

const [worktree, out] = process.argv.slice(2) as [string, string];
const nightly = join(worktree, "scripts", "nightly");
const { apiValues, routesOfValues } = await import(join(nightly, "api.ts"));
const { rootValues, issuingValues } = await import(join(nightly, "certs.ts"));
const { discoverShapes } = await import(join(nightly, "smoke.ts"));

const chartDir = join(worktree, "deploy", "chart");
mkdirSync(out, { recursive: true });

// Throwaway certificates only for the render: the keys are removed right after.
const keys = mkdtempSync(join(out, "keys-"));
const sh = (cmd: string[]) => {
  const r = Bun.spawnSync(cmd, { cwd: keys, stdout: "pipe", stderr: "pipe" });
  if (r.exitCode !== 0) throw new Error(`${cmd.join(" ")}: ${r.stderr.toString()}`);
  return r.stdout.toString();
};
let root: string;
let issuing: string;
try {
  sh(["openssl", "req", "-x509", "-newkey", "ec", "-pkeyopt", "ec_paramgen_curve:P-384", "-nodes", "-subj", "/CN=render root", "-days", "1", "-keyout", "root.key", "-out", "root.pem"]);
  sh(["openssl", "req", "-new", "-newkey", "ec", "-pkeyopt", "ec_paramgen_curve:P-256", "-nodes", "-subj", "/CN=render issuing", "-keyout", "issuing.key", "-out", "issuing.csr"]);
  sh(["openssl", "x509", "-req", "-in", "issuing.csr", "-CA", "root.pem", "-CAkey", "root.key", "-set_serial", "1", "-days", "1", "-out", "issuing.pem"]);
  root = readFileSync(join(keys, "root.pem"), "utf8");
  issuing = readFileSync(join(keys, "issuing.pem"), "utf8");
} finally {
  rmSync(keys, { recursive: true, force: true });
}

const run = (cmd: string[], input?: string) => {
  const r = Bun.spawnSync(cmd, { cwd: chartDir, stdin: input === undefined ? "ignore" : new TextEncoder().encode(input), stdout: "pipe", stderr: "pipe" });
  return { code: r.exitCode, stdout: r.stdout.toString(), stderr: r.stderr.toString() };
};
const { pinnedTool } = await import(join(nightly, "exec.ts"));
const helm: string = await pinnedTool("helm", worktree);
const kubeconform: string = await pinnedTool("kubeconform", worktree);
const cache = join(out, "kubeconform-cache");
mkdirSync(cache, { recursive: true });
const CRDS = "https://raw.githubusercontent.com/datreeio/CRDs-catalog/main/{{.Group}}/{{.ResourceKind}}_{{.ResourceAPIVersion}}.json";

let failed = false;
for (const shape of discoverShapes(chartDir)) {
  const files = shape.valuesFiles.map((f: string) => join(chartDir, f));
  const api = join(out, `values-api-${shape.name}.yaml`);
  writeFileSync(api, Bun.YAML.stringify(apiValues(shape.name, routesOfValues(chartDir, shape.valuesFiles)), null, 2));
  const certs = join(out, `values-certs-${shape.name}.yaml`);
  writeFileSync(certs, Bun.YAML.stringify(rootValues(root), null, 2));
  const certsIssuing = join(out, `values-certs-issuing-${shape.name}.yaml`);
  writeFileSync(certsIssuing, Bun.YAML.stringify(issuingValues(issuing), null, 2));
  for (const [stage, extra] of [["waiting", [api, certs]], ["signing", [api, certs, certsIssuing]]] as const) {
    const args = [...files, ...extra].flatMap((f) => ["-f", f]);
    const rendered = run([helm, "template", "trustiot", ".", "--namespace", `nightly-${shape.name}`, ...args]);
    if (rendered.code !== 0) {
      console.log(`${shape.name} ${stage}: helm template failed: ${rendered.stderr}`);
      failed = true;
      continue;
    }
    const certsDocs = rendered.stdout.split("\n---").filter((d) => /name: trustiot-certs/.test(d)).length;
    for (const version of ["1.30.0", "1.36.0"]) {
      const checked = run([kubeconform, "-strict", "-summary", "-kubernetes-version", version, "-cache", cache, "-schema-location", "default", "-schema-location", CRDS, "-"], rendered.stdout);
      console.log(`${shape.name} ${stage} k8s ${version}: kubeconform exit ${checked.code}; ${checked.stdout.trim().split("\n").at(-1)}; documents naming trustiot-certs*: ${certsDocs}`);
      if (checked.code !== 0) {
        console.log(checked.stdout);
        failed = true;
      }
    }
  }
}
process.exit(failed ? 1 : 0);
