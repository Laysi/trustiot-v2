fn describe(error: &sqlx::Error) -> String {
    match error.as_database_error() {
        Some(db) => format!(
            "{} {}",
            db.code().map(|c| c.into_owned()).unwrap_or_default(),
            db.message()
        ),
        None => format!("non-database error: {error}"),
    }
}

fn sqlstate(error: &sqlx::Error) -> Option<String> {
    error
        .as_database_error()
        .and_then(|e| e.code())
        .map(|c| c.into_owned())
}

/// Runs `stmt` in its own transaction after `setup`; commits on success, rolls back on error.
async fn attempt(conn: &mut PgConnection, setup: &str, stmt: &str) -> Result<u64, String> {
    sqlx::raw_sql("BEGIN").execute(&mut *conn).await.expect("begin");
    if !setup.is_empty() {
        sqlx::raw_sql(AssertSqlSafe(setup.to_owned()))
            .execute(&mut *conn)
            .await
            .expect("setup");
    }
    match sqlx::raw_sql(AssertSqlSafe(stmt.to_owned()))
        .execute(&mut *conn)
        .await
    {
        Ok(done) => {
            sqlx::raw_sql("COMMIT")
                .execute(&mut *conn)
                .await
                .expect("commit");
            Ok(done.rows_affected())
        }
        Err(error) => {
            sqlx::raw_sql("ROLLBACK")
                .execute(&mut *conn)
                .await
                .expect("rollback");
            Err(describe(&error))
        }
    }
}

async fn run_list(
    conn: &mut PgConnection,
    label: &str,
    setup: &str,
    list: &[(&'static str, String)],
) -> Vec<(String, Result<u64, String>)> {
    let mut out = Vec::new();
    for (case, stmt) in list {
        let result = attempt(conn, setup, stmt).await;
        println!("[{label}] {case}: {result:?}");
        out.push(((*case).to_owned(), result));
    }
    out
}

/// Probes 1 and 3: every write path and role escape of the producer's platform role.
#[tokio::test]
async fn probe_platform_role_tenant_writes_and_escapes() {
    let Some(fixture) = Fixture::new("v360_writes", template("example_reader")).await else {
        return;
    };
    let before = fixture.outbox_counts().await;
    println!("owner view before: (tenant rows, all rows) = {before:?}");

    let mut platform = fixture.connect("example_platform").await;
    let privileges: (bool, bool, bool, String, String) = sqlx::query_as(
        "SELECT has_schema_privilege('public', 'CREATE'), \
                has_database_privilege(current_database(), 'CREATE'), \
                has_database_privilege(current_database(), 'TEMP'), \
                (SELECT string_agg(p, ',') FROM unnest(ARRAY['SELECT','INSERT','UPDATE','DELETE', \
                    'TRUNCATE','REFERENCES','TRIGGER']) AS p \
                 WHERE has_table_privilege('outbox_event', p)), \
                coalesce((SELECT string_agg(nspname, ',') FROM pg_namespace \
                 WHERE has_schema_privilege(oid, 'CREATE')), '')",
    )
    .fetch_one(&mut platform)
    .await
    .expect("privileges");
    println!(
        "example_platform: CREATE on public={}, CREATE on database={}, TEMP={}, \
         privileges on outbox_event=[{}], schemas with CREATE=[{}]",
        privileges.0, privileges.1, privileges.2, privileges.3, privileges.4
    );
    let memberships: Vec<(String, String, bool, bool, bool)> = sqlx::query_as(
        "SELECT roleid::regrole::text, member::regrole::text, admin_option, inherit_option, \
                set_option FROM pg_auth_members \
         WHERE member::regrole::text LIKE 'example%' OR roleid::regrole::text LIKE 'example%' \
         ORDER BY 1, 2",
    )
    .fetch_all(&mut platform)
    .await
    .expect("memberships");
    println!("memberships (role, member, admin, inherit, set): {memberships:?}");
    let member_of: Vec<(String, bool, bool)> = sqlx::query_as(
        "SELECT r, pg_has_role('example_platform', r, 'MEMBER'), \
                pg_has_role('example_platform', r, 'SET') \
         FROM unnest(ARRAY['example_app','example_owner','example_reader','example_platform_group']) r",
    )
    .fetch_all(&mut platform)
    .await
    .expect("member of");
    println!("example_platform member of / may SET ROLE to: {member_of:?}");

    let writes = tenant_write_statements();
    let tx_setup = format!("SELECT set_config('trustiot.tenant_id', '{TENANT_A}', true)");
    let mut results = run_list(&mut platform, "tx tenant", &tx_setup, &writes).await;
    sqlx::raw_sql(AssertSqlSafe(format!(
        "SELECT set_config('trustiot.tenant_id', '{TENANT_A}', false)"
    )))
    .execute(&mut platform)
    .await
    .expect("session tenant");
    results.extend(run_list(&mut platform, "session tenant", "", &writes).await);

    // Extended protocol with a bound parameter (sqlx prepares it), session tenant still set.
    let bound = sqlx::query("INSERT INTO outbox_event (tenant_id, envelope) VALUES ($1::uuid, '{}')")
        .bind(TENANT_A)
        .execute(&mut platform)
        .await;
    println!(
        "[session tenant] extended protocol, bound tenant: {:?}",
        bound.as_ref().map(|d| d.rows_affected()).map_err(describe)
    );
    assert_eq!(bound.as_ref().err().and_then(sqlstate).as_deref(), Some("42501"));

    // COPY FROM STDIN, session tenant set.
    let copy_result = match platform
        .copy_in_raw("COPY outbox_event (tenant_id, envelope) FROM STDIN")
        .await
    {
        Ok(mut copy) => {
            copy.send(format!("{TENANT_A}\t{{}}\n").into_bytes())
                .await
                .expect("send");
            copy.finish().await.map_err(|e| describe(&e))
        }
        Err(error) => Err(describe(&error)),
    };
    println!("[session tenant] COPY FROM STDIN: {copy_result:?}");
    assert!(copy_result.is_err(), "COPY must be refused");
    results.push(("COPY FROM STDIN".to_owned(), copy_result));

    let objects = object_statements();
    results.extend(run_list(&mut platform, "objects", "", &objects).await);
    let roles = role_statements();
    results.extend(run_list(&mut platform, "roles (session tenant)", "", &roles).await);
    drop(platform);
    // Role escapes again on a fresh session with the tenant only in the transaction.
    let mut platform = fixture.connect("example_platform").await;
    results.extend(run_list(&mut platform, "roles (tx tenant)", &tx_setup, &roles).await);

    let after = fixture.outbox_counts().await;
    println!("owner view after: (tenant rows, all rows) = {after:?}");
    assert_eq!(after.0, before.0, "no tenant event landed");
    let accepted: Vec<_> = results.iter().filter(|(_, r)| r.is_ok()).collect();
    println!("accepted statements (none may have written a tenant row): {accepted:?}");
    for (case, result) in &results {
        if let Err(text) = result {
            assert!(
                text.starts_with("42501") || text.starts_with("0A000"),
                "{case}: unexpected error {text}"
            );
        }
    }
}

/// Probe 2: the producer's platform role is also the consumer reader role.
#[tokio::test]
async fn probe_self_consuming_platform_role() {
    let Some(fixture) = Fixture::new("v360_self", template("example_platform")).await else {
        return;
    };
    let before = fixture.outbox_counts().await;
    let mut platform = fixture.connect("example_platform").await;
    let tx_setup = format!("SELECT set_config('trustiot.tenant_id', '{TENANT_A}', true)");
    let mut results =
        run_list(&mut platform, "self tx tenant", &tx_setup, &tenant_write_statements()).await;
    sqlx::raw_sql(AssertSqlSafe(format!(
        "SELECT set_config('trustiot.tenant_id', '{TENANT_B}', false)"
    )))
    .execute(&mut platform)
    .await
    .expect("session tenant");
    let b_insert = vec![(
        "single VALUES tenant B, session tenant B",
        format!("INSERT INTO outbox_event (tenant_id, envelope) VALUES ('{TENANT_B}', '{{}}') RETURNING tenant_id"),
    )];
    results.extend(run_list(&mut platform, "self session tenant", "", &b_insert).await);
    let after = fixture.outbox_counts().await;
    assert_eq!(after.0, before.0, "no tenant event landed (self-consuming)");
    for (case, result) in &results {
        if let Err(text) = result {
            assert!(text.starts_with("42501") || text.starts_with("0A000"), "{case}: {text}");
        }
    }
    drop(platform);

    // Platform-wide writes still work, reads see every event, the checks accept it.
    let db = fixture.platform_db("example_platform");
    let actor = PlatformActor::service(ServiceIdentity::new(ServiceName::Identity));
    let mut tx = db.begin(actor, "verify-360").await.expect("begin");
    let returned: Option<String> = sqlx::query_scalar(
        "INSERT INTO outbox_event (tenant_id, envelope) VALUES (NULL, '{}') RETURNING tenant_id::text",
    )
    .fetch_one(&mut *tx)
    .await
    .expect("a platform-wide event, with RETURNING (it has SELECT now)");
    assert_eq!(returned, None);
    tx.commit().await.expect("commit");
    let reader = db.reader(actor, "outbox-consumer-read");
    let mut tx = reader.begin().await.expect("reader begin");
    let seen: Vec<Option<String>> =
        sqlx::query_scalar("SELECT tenant_id::text FROM outbox_event ORDER BY id")
            .fetch_all(&mut *tx)
            .await
            .expect("read");
    let reads_all: Option<bool> = sqlx::query_scalar(READS_ALL_TENANTS)
        .fetch_optional(&mut *tx)
        .await
        .expect("reader check");
    tx.rollback().await.expect("rollback");
    println!("self-consuming reader sees {seen:?}; READS_ALL_TENANTS = {reads_all:?}");
    assert_eq!(seen, [None, Some(TENANT_A.to_owned()), None]);
    assert_eq!(reads_all, Some(true));
    for (role, as_role) in [
        ("example_platform", SessionRole::Platform),
        ("example_app", SessionRole::Service),
    ] {
        let mut session = fixture.connect(role).await;
        let checked = postgres::check_session(&mut session, as_role).await.expect("check");
        println!("self-consuming: check_session({role}, {as_role:?}) = {checked:?}");
        assert_eq!(checked, Ok(()));
    }
}

/// Probe 4 and the reader check in the normal (separate reader) layout; tenant-table UPDATE.
#[tokio::test]
async fn probe_legitimate_paths_and_reader_checks() {
    let Some(fixture) = Fixture::new("v360_legit", template("example_reader")).await else {
        return;
    };
    let mut app = fixture.connect("example_app").await;
    let set_a = format!("SELECT set_config('trustiot.tenant_id', '{TENANT_A}', true)");
    for (case, stmt) in [
        ("tenant A", format!("INSERT INTO outbox_event (tenant_id, envelope) VALUES ('{TENANT_A}', '{{}}')")),
        ("tenant B", format!("INSERT INTO outbox_event (tenant_id, envelope) VALUES ('{TENANT_B}', '{{}}')")),
        ("platform-wide", "INSERT INTO outbox_event (tenant_id, envelope) VALUES (NULL, '{}')".to_owned()),
    ] {
        let result = attempt(&mut app, &set_a, &stmt).await;
        println!("[service role, tenant A] {case}: {result:?}");
        match case {
            "tenant A" => assert_eq!(result, Ok(1)),
            _ => assert!(result.unwrap_err().starts_with("42501")),
        }
    }
    let mut platform = fixture.connect("example_platform").await;
    let result = attempt(&mut platform, "", "INSERT INTO outbox_event (tenant_id, envelope) VALUES (NULL, '{}')").await;
    println!("[platform role, no tenant] platform-wide: {result:?}");
    assert_eq!(result, Ok(1));
    let result = attempt(
        &mut platform,
        "",
        &format!("INSERT INTO outbox_event (tenant_id, envelope) VALUES ('{TENANT_A}', '{{}}')"),
    )
    .await;
    println!("[platform role, no tenant] tenant A: {result:?}");
    assert!(result.unwrap_err().starts_with("42501"));

    for (role, expected) in [("example_reader", Some(true)), ("example_platform", Some(false))] {
        let mut session = fixture.connect(role).await;
        let reads_all: Option<bool> = sqlx::query_scalar(READS_ALL_TENANTS)
            .fetch_optional(&mut session)
            .await
            .expect("reader check");
        println!("READS_ALL_TENANTS as {role}: {reads_all:?}");
        assert_eq!(reads_all, expected);
    }
    let mut reader = fixture.connect("example_reader").await;
    let seen: Vec<Option<String>> =
        sqlx::query_scalar("SELECT tenant_id::text FROM outbox_event ORDER BY id")
            .fetch_all(&mut reader)
            .await
            .expect("read");
    println!("reader sees {seen:?}");
    assert_eq!(seen, [None, Some(TENANT_A.to_owned()), Some(TENANT_A.to_owned()), None]);
    for (role, as_role) in [
        ("example_app", SessionRole::Service),
        ("example_platform", SessionRole::Platform),
        ("example_reader", SessionRole::Platform),
    ] {
        let mut session = fixture.connect(role).await;
        let checked = postgres::check_session(&mut session, as_role).await.expect("check");
        println!("check_session({role}, {as_role:?}) = {checked:?}");
        assert_eq!(checked, Ok(()));
    }

    // tenant-table.sql, platform_access: UPDATE with and without a self-set tenant.
    let without = attempt(&mut platform, "", "UPDATE example_items SET name = name").await;
    let with = attempt(&mut platform, &set_a, "UPDATE example_items SET name = name").await;
    let moved = attempt(
        &mut platform,
        &set_a,
        &format!("UPDATE example_items SET tenant_id = '{TENANT_B}' WHERE name = 'zzz-none'"),
    )
    .await;
    println!(
        "platform_access UPDATE rows: without tenant {without:?}, with tenant A {with:?}; \
         (no-op move) {moved:?}"
    );
    assert_eq!(without, with);
}
