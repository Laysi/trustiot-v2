```sh
  --certificate-oidc-issuer https://token.actions.githubusercontent.com

  --certificate-identity 'https://github.com/Laysi/trustiot-v2/.github/workflows/release.yml@refs/tags/v0.0.0-rc.1' \
  --certificate-oidc-issuer https://token.actions.githubusercontent.com
# SBOM：驗證 attestation 後取出 predicate
cosign verify-attestation --type spdxjson ghcr.io/laysi/trustiot-v2/release-selftest:0.0.0-rc.1 \
  --certificate-identity 'https://github.com/Laysi/trustiot-v2/.github/workflows/release.yml@refs/tags/v0.0.0-rc.1' \

```
