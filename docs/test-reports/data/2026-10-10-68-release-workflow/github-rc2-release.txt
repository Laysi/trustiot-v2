  --certificate-identity 'https://github.com/Laysi/trustiot-v2/.github/workflows/release.yml@refs/tags/v0.0.0-rc.2' \

cosign verify-attestation --type spdxjson ghcr.io/laysi/trustiot-v2/release-selftest:0.0.0-rc.2 \
  --certificate-identity 'https://github.com/Laysi/trustiot-v2/.github/workflows/release.yml@refs/tags/v0.0.0-rc.2' \
